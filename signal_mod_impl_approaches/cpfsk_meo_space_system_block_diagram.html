CPFSK Mathematical Definition
Transmitted Signal:
s(t) = A * cos(2*pi*f_c*t + phi(t))
Instantaneous Phase:
phi(t) = 2*pi*h * sum(d_k * q(t - kT))
Phase Pulse Shape (Rectangular):
q(t) = t/(2T) for 0 <= t < T
q(t) = 1/2 for t >= T

h = modulation index (0.5 for MSK)
d_k = +/-1 (bipolar data symbols)
T = symbol period
Frequency Deviation:
delta_f = h / (2*T) = h * R_b / 2
Phase Change per Symbol:
delta_phi = +/- h * pi radians

# ============== RECTANGULAR PULSE DESIGN ==============

def rectangular_pulse(sps, num_symbols=1):
    """
    Design rectangular frequency pulse for CPFSK modulation
    sps: samples per symbol
    num_symbols: pulse duration in symbols (typically 1 for CPFSK)

    Returns: rectangular pulse filter
    """
    # Rectangular pulse of duration T (1 symbol)
    num_taps = num_symbols * sps
    h = np.ones(num_taps) / sps

    return h

def phase_pulse_rect(sps):
    """
    Generate rectangular phase pulse shape q(t) for CPFSK
    q(t) is the integral of the frequency pulse
    For rectangular: q(t) = t/(2T) for 0 <= t < T, then 1/2
    """
    # For one symbol duration
    t = np.arange(sps) / sps  # 0 to 1 in normalized time
    q = t / 2  # Linear ramp from 0 to 0.5

    return q

# ============== CPFSK MODULATION ==============

def cpfsk_modulate(bits, sps=SAMPLES_PER_SYMBOL, h=MOD_INDEX):
    """
    CPFSK modulation (Continuous Phase FSK)
    bits: input binary data (0/1)
    sps: samples per symbol
    h: modulation index (0.5 = MSK)

    Returns: complex baseband CPFSK signal (constant envelope)
    """
    # Convert bits to bipolar NRZ (+1/-1)
    nrz = 2 * bits.astype(float) - 1

    # Upsample by inserting zeros
    upsampled = np.zeros(len(nrz) * sps)
    upsampled[::sps] = nrz

    # Apply rectangular pulse filter
    rect = rectangular_pulse(sps)
    freq_pulse = np.convolve(upsampled, rect, mode='full')[:len(upsampled)]

    # Scale by modulation index h and pi
    # Phase change per symbol = +/- h*pi
    freq_scaled = freq_pulse * h * np.pi

    # Integrate to get continuous phase
    phase = np.cumsum(freq_scaled)

    # Generate complex baseband signal (constant envelope)
    signal_out = np.exp(1j * phase)

    return signal_out, phase, rect

def cpfsk_modulate_direct(bits, sps=SAMPLES_PER_SYMBOL, h=MOD_INDEX):
    """
    Direct CPFSK modulation using phase accumulation
    More intuitive implementation showing phase tree structure
    """
    num_samples = len(bits) * sps
    phase = np.zeros(num_samples)

    # Phase increment per sample for each symbol
    delta_phi_per_sample = h * np.pi / sps

    current_phase = 0

    for i, bit in enumerate(bits):
        # Determine phase direction based on bit
        direction = 1 if bit == 1 else -1

        # Generate phase for this symbol (linear ramp)
        for j in range(sps):
            sample_idx = i * sps + j
            current_phase += direction * delta_phi_per_sample
            phase[sample_idx] = current_phase

    # Generate complex signal
    signal_out = np.exp(1j * phase)

    return signal_out, phase

def cpfsk_demodulate(signal_in, sps=SAMPLES_PER_SYMBOL):
    """
    CPFSK demodulation using FM discriminator (non-coherent)

    Returns: recovered bits
    """
    # FM discriminator: differentiate phase
    phase = np.unwrap(np.angle(signal_in))
    freq = np.diff(phase)
    freq = np.append(freq, freq[-1])  # Pad to same length

    # Apply matched rectangular filter (integrate and dump)
    rect = rectangular_pulse(sps)
    filtered = np.convolve(freq, rect, mode='same')

    # Downsample at symbol centers
    symbols = filtered[sps//2::sps]

    # Hard decision (sign detection)
    bits = (symbols > 0).astype(int)

    return bits, symbols

def cpfsk_coherent_demod(signal_in, sps=SAMPLES_PER_SYMBOL, h=MOD_INDEX):
    """
    Coherent CPFSK demodulation using matched filter bank
    Optimal for h=0.5 (MSK) - uses I/Q decomposition
    """
    # For MSK (h=0.5), use I/Q matched filters
    # I channel: cos(pi*t/2T), Q channel: sin(pi*t/2T)

    t = np.arange(sps) / sps  # Normalized time 0 to 1

    # Matched filter for I arm (even symbols)
    mf_i = np.cos(np.pi * h * t)

    # Matched filter for Q arm (odd symbols)
    mf_q = np.sin(np.pi * h * t)

    # Get I and Q components
    sig_i = np.real(signal_in)
    sig_q = np.imag(signal_in)

    # Correlate with matched filters
    corr_i = np.correlate(sig_i, mf_i, mode='same')
    corr_q = np.correlate(sig_q, mf_q, mode='same')

    # Sample at symbol boundaries
    # For MSK: even bits from I, odd bits from Q (offset by T/2)
    samples_i = corr_i[sps//2::sps]  # Even symbols
    samples_q = corr_q[sps::sps]       # Odd symbols (offset)

    # Interleave decisions
    num_symbols = min(len(samples_i), len(samples_q)) * 2
    bits = np.zeros(num_symbols, dtype=int)
    bits[0::2] = (samples_i[:num_symbols//2] > 0).astype(int)
    bits[1::2] = (samples_q[:num_symbols//2] > 0).astype(int)

    return bits

# ============== FM DISCRIMINATOR ==============

def fm_discriminator_delay_multiply(signal_in, delay_samples=1):
    """
    FM discriminator using delay and multiply method
    More robust to noise than direct differentiation
    """
    # Delay signal
    delayed = np.roll(signal_in, delay_samples)

    # Multiply by conjugate of delayed signal
    product = signal_in * np.conj(delayed)

    # Extract instantaneous frequency (phase difference)
    freq_out = np.angle(product)

    return freq_out

def integrate_and_dump(signal_in, sps):
    """
    Integrate and dump filter (matched filter for rectangular pulses)
    Optimal for CPFSK demodulation
    """
    # Reshape into symbol-length segments
    num_symbols = len(signal_in) // sps
    truncated = signal_in[:num_symbols * sps]
    reshaped = truncated.reshape(num_symbols, sps)

    # Integrate (sum) each segment
    output = np.sum(reshaped, axis=1)

    return output

# ============== CONVOLUTIONAL ENCODER ==============

def conv_encode(bits, g1=0o171, g2=0o133):
    """
    Convolutional encoder - NASA standard rate 1/2, K=7
    """
    K = 7
    state = 0
    encoded = []

    for bit in bits:
        state = ((state << 1) | bit) & (2**K - 1)
        out1 = bin(state & g1).count('1') % 2
        out2 = bin(state & g2).count('1') % 2
        encoded.extend([out1, out2])

    # Flush encoder
    for _ in range(K - 1):
        state = (state << 1) & (2**K - 1)
        out1 = bin(state & g1).count('1') % 2
        out2 = bin(state & g2).count('1') % 2
        encoded.extend([out1, out2])

    return np.array(encoded)

# ============== INTERLEAVER ==============

def block_interleave(bits, rows=32):
    """Block interleaver for burst error protection"""
    cols = int(np.ceil(len(bits) / rows))
    padded = np.zeros(rows * cols, dtype=int)
    padded[:len(bits)] = bits
    matrix = padded.reshape(rows, cols)
    return matrix.T.flatten()

def block_deinterleave(bits, rows=32):
    """Block de-interleaver"""
    cols = int(np.ceil(len(bits) / rows))
    matrix = bits[:rows*cols].reshape(cols, rows)
    return matrix.T.flatten()

# ============== SCRAMBLER ==============

def lfsr_sequence(length, seed=0x1ACFFC1D):
    """Generate LFSR pseudo-random sequence"""
    state = seed
    seq = np.zeros(length, dtype=int)
    for i in range(length):
        seq[i] = state & 1
        newbit = ((state >> 0) ^ (state >> 2) ^ (state >> 3) ^ (state >> 5)) & 1
        state = (state >> 1) | (newbit << 31)
    return seq

def scramble(bits):
    pn = lfsr_sequence(len(bits))
    return (bits ^ pn).astype(int)

def descramble(bits):
    return scramble(bits)

# ============== SPACE CHANNEL MODEL ==============

def free_space_path_loss_db(distance, freq=CARRIER_FREQ):
    """Calculate FSPL in dB"""
    fspl = (4 * np.pi * distance * freq / c) ** 2
    return 10 * np.log10(fspl)

def apply_doppler_shift(signal_in, doppler_hz, sample_rate):
    """Apply Doppler frequency shift"""
    t = np.arange(len(signal_in)) / sample_rate
    shift = np.exp(2j * np.pi * doppler_hz * t)
    return signal_in * shift

def apply_frequency_drift(signal_in, drift_rate_hz_per_s, sample_rate):
    """Apply slow frequency drift (oscillator aging/thermal)"""
    t = np.arange(len(signal_in)) / sample_rate
    phase = 2 * np.pi * drift_rate_hz_per_s * t**2 / 2
    return signal_in * np.exp(1j * phase)

def meo_space_channel(signal_in, distance_km, doppler_hz, snr_db):
    """Complete MEO space channel model for CPFSK (baseband)"""
    # Apply Doppler shift
    sig = apply_doppler_shift(signal_in, doppler_hz, SAMPLE_RATE)

    # Apply small frequency drift
    sig = apply_frequency_drift(sig, 10, SAMPLE_RATE)  # 10 Hz/s drift

    # Add AWGN
    signal_power = np.mean(np.abs(sig) ** 2)
    snr_linear = 10 ** (snr_db / 10)
    noise_power = signal_power / snr_linear
    noise = np.sqrt(noise_power/2) * (np.random.randn(len(sig)) + 1j*np.random.randn(len(sig)))

    return sig + noise

# ============== RF UP-CONVERSION (TRANSMITTER) ==============

def upsample_for_rf(baseband_signal, oversample_factor=RF_OVERSAMPLE):
    """Upsample baseband signal for RF processing (80 MHz -> 320 MHz)"""
    # Zero-insertion upsampling
    upsampled = np.zeros(len(baseband_signal) * oversample_factor, dtype=complex)
    upsampled[::oversample_factor] = baseband_signal

    # Anti-imaging lowpass filter
    nyq = 0.5 * oversample_factor
    cutoff = 0.4 / nyq
    b, a = butter(6, cutoff, btype='low')
    upsampled_i = filtfilt(b, a, np.real(upsampled)) * oversample_factor
    upsampled_q = filtfilt(b, a, np.imag(upsampled)) * oversample_factor

    return upsampled_i + 1j * upsampled_q

def upconvert_to_carrier(baseband_signal, if_freq=IF_FREQ, sample_rate=RF_SAMPLE_RATE):
    """Upconvert baseband to IF frequency (represents 2.2 GHz carrier)"""
    upsampled = upsample_for_rf(baseband_signal)
    t = np.arange(len(upsampled)) / sample_rate
    carrier_i = np.cos(2 * np.pi * if_freq * t)
    carrier_q = -np.sin(2 * np.pi * if_freq * t)
    rf_signal = np.real(upsampled) * carrier_i + np.imag(upsampled) * carrier_q
    return rf_signal, t

def tx_power_amplifier(rf_signal, tx_power_w=TX_POWER_W):
    """SSPA model - constant envelope allows saturated operation!"""
    signal_rms = np.sqrt(np.mean(rf_signal**2))
    target_rms = np.sqrt(tx_power_w)
    return rf_signal * (target_rms / signal_rms)

# ============== RF CHANNEL MODEL (CARRIER FREQUENCY) ==============

def rf_free_space_path_loss(distance_m, freq=CARRIER_FREQ):
    """Calculate FSPL at carrier frequency"""
    fspl_linear = (4 * np.pi * distance_m * freq / c) ** 2
    return fspl_linear

def rf_doppler_shift(rf_signal, doppler_hz, sample_rate=RF_SAMPLE_RATE):
    """Apply Doppler shift to RF signal (max ~22 kHz @ 2.2 GHz)"""
    t = np.arange(len(rf_signal)) / sample_rate
    analytic = signal.hilbert(rf_signal)
    shifted = analytic * np.exp(2j * np.pi * doppler_hz * t)
    return np.real(shifted)

def rf_oscillator_drift(rf_signal, drift_rate_hz_per_s, sample_rate=RF_SAMPLE_RATE):
    """Apply frequency drift due to oscillator instability (10 Hz/s typical)"""
    t = np.arange(len(rf_signal)) / sample_rate
    analytic = signal.hilbert(rf_signal)
    drift_phase = 2 * np.pi * drift_rate_hz_per_s * t**2 / 2
    drifted = analytic * np.exp(1j * drift_phase)
    return np.real(drifted)

def rf_awgn_channel(rf_signal, snr_db, sample_rate=RF_SAMPLE_RATE):
    """Add AWGN to RF signal"""
    signal_power = np.mean(rf_signal**2)
    snr_linear = 10 ** (snr_db / 10)
    noise_power = signal_power / snr_linear
    noise = np.sqrt(noise_power) * np.random.randn(len(rf_signal))
    return rf_signal + noise

def meo_rf_channel(rf_signal, distance_km, doppler_hz, eb_n0_db):
    """Complete MEO RF channel model at carrier frequency"""
    sig = rf_doppler_shift(rf_signal, doppler_hz)
    sig = rf_oscillator_drift(sig, 10)  # 10 Hz/s drift
    sig = rf_awgn_channel(sig, eb_n0_db)
    return sig

# ============== RF DOWN-CONVERSION (RECEIVER FRONT-END) ==============

def rx_low_noise_amplifier(rf_signal, noise_figure_db=2.0):
    """Model receiver LNA with noise figure"""
    nf_linear = 10 ** (noise_figure_db / 10)
    signal_power = np.mean(rf_signal**2)
    added_noise_power = signal_power * (nf_linear - 1) / 100
    noise = np.sqrt(added_noise_power) * np.random.randn(len(rf_signal))
    return rf_signal + noise

def downconvert_to_baseband(rf_signal, if_freq=IF_FREQ, sample_rate=RF_SAMPLE_RATE):
    """Downconvert RF to complex baseband (320 MHz -> 80 MHz)"""
    t = np.arange(len(rf_signal)) / sample_rate
    lo_i = 2 * np.cos(2 * np.pi * if_freq * t)
    lo_q = -2 * np.sin(2 * np.pi * if_freq * t)
    i_mixed = rf_signal * lo_i
    q_mixed = rf_signal * lo_q

    # Lowpass filter to remove 2*IF component
    nyq = 0.5 * sample_rate
    cutoff = (BASEBAND_SAMPLE_RATE / 2) / nyq * 0.8
    b, a = butter(6, cutoff, btype='low')
    i_baseband = filtfilt(b, a, i_mixed)
    q_baseband = filtfilt(b, a, q_mixed)

    # Decimate to baseband sample rate (4x)
    decimate_factor = int(sample_rate / BASEBAND_SAMPLE_RATE)
    i_decimated = i_baseband[::decimate_factor]
    q_decimated = q_baseband[::decimate_factor]
    return i_decimated + 1j * q_decimated

def rx_agc(baseband_signal, target_power=1.0):
    """Automatic Gain Control - normalize signal power"""
    current_power = np.mean(np.abs(baseband_signal)**2)
    if current_power > 0:
        gain = np.sqrt(target_power / current_power)
        return baseband_signal * gain
    return baseband_signal

# ============== AFC (Automatic Frequency Control) ==============

def afc_estimate(signal_in, sps, sample_rate=BASEBAND_SAMPLE_RATE):
    """
    Estimate frequency offset using differential phase method.
    For CPFSK/MSK, FFT peak detection doesn't work - uses average phase rotation.
    """
    # Method: Estimate frequency from average phase rotation
    delay = sps  # Average over one symbol period for noise reduction

    # Compute conjugate product for phase difference
    product = signal_in[delay:] * np.conj(signal_in[:-delay])

    # Average the phase difference
    avg_product = np.mean(product)
    phase_diff = np.angle(avg_product)

    # Frequency = phase_diff / (2*pi*delay_time)
    delay_time = delay / sample_rate
    freq_offset = phase_diff / (2 * np.pi * delay_time)

    # Apply threshold - if estimated offset is very small, treat as zero
    if np.abs(freq_offset) < 50:  # 50 Hz threshold
        freq_offset = 0

    return freq_offset

# ============== TRANSMITTER ==============

def meo_cpfsk_transmitter(data_bits, include_rf_upconversion=True):
    """
    Complete MEO CPFSK satellite transmitter chain with RF front-end
    Processing: FEC -> Interleave -> Scramble -> CPFSK Mod -> RF Upconv -> SSPA
    """
    # Convolutional encoding (rate 1/2)
    encoded = conv_encode(data_bits)

    # Interleaving
    interleaved = block_interleave(encoded)

    # Scrambling
    scrambled = scramble(interleaved)

    # CPFSK modulation (generates complex baseband signal)
    baseband_signal, phase, rect = cpfsk_modulate(scrambled)

    if include_rf_upconversion:
        # Up-convert baseband to carrier frequency (2.2 GHz via IF)
        rf_signal, t_rf = upconvert_to_carrier(baseband_signal)
        # Power amplifier (SSPA in saturation - OK for constant envelope!)
        rf_signal = tx_power_amplifier(rf_signal)
        return rf_signal, baseband_signal, scrambled, rect
    else:
        return baseband_signal, scrambled, rect

# ============== RECEIVER ==============

def meo_cpfsk_receiver(rx_signal, num_coded_bits, num_info_bits, doppler_hz=0, is_rf_signal=True):
    """
    Complete MEO CPFSK satellite receiver chain with RF front-end
    Processing: LNA -> Downconv -> AGC -> Doppler Comp -> AFC -> FM Demod -> FEC Decode
    """
    if is_rf_signal:
        # === RF FRONT-END PROCESSING ===
        rx_rf = rx_low_noise_amplifier(rx_signal)  # LNA
        rx_baseband = downconvert_to_baseband(rx_rf)  # Down-convert to baseband
        rx_baseband = rx_agc(rx_baseband)  # AGC normalize
    else:
        rx_baseband = rx_signal  # Input is already baseband

    # === BASEBAND PROCESSING ===
    # Doppler compensation (coarse)
    if doppler_hz != 0:
        rx_baseband = apply_doppler_shift(rx_baseband, -doppler_hz, BASEBAND_SAMPLE_RATE)

    # AFC fine tuning
    freq_offset = afc_estimate(rx_baseband, SAMPLES_PER_SYMBOL)
    rx_baseband = apply_doppler_shift(rx_baseband, -freq_offset, BASEBAND_SAMPLE_RATE)

    # CPFSK demodulation (non-coherent FM discriminator)
    rx_bits, symbols = cpfsk_demodulate(rx_baseband)

    # Trim to expected length
    rx_bits = rx_bits[:num_coded_bits]

    # Descramble
    descrambled = descramble(rx_bits)

    # De-interleave
    deinterleaved = block_deinterleave(descrambled)

    # Viterbi decode
    decoded = viterbi_decode_hard(deinterleaved, num_info_bits)

    return decoded[:num_info_bits], symbols, rx_baseband

# ============== LINK BUDGET ==============

def calculate_link_budget(distance_km):
    """Calculate link budget for CPFSK MEO ISL"""
    distance = distance_km * 1000

    tx_power_dbw = 10 * np.log10(TX_POWER_W)
    # No back-off needed for constant envelope!
    eirp = tx_power_dbw + TX_ANTENNA_GAIN_DBI - LINE_LOSS_DB
    fspl = free_space_path_loss_db(distance)
    rx_gain = RX_ANTENNA_GAIN_DBI - LINE_LOSS_DB
    g_over_t = rx_gain - 10 * np.log10(SYSTEM_NOISE_TEMP_K)
    c_n0 = eirp - fspl - 0.5 + g_over_t + 228.6  # Boltzmann
    eb_n0 = c_n0 - 10 * np.log10(SYMBOL_RATE)

    return {
        'EIRP_dBW': eirp,
        'FSPL_dB': fspl,
        'C_N0_dBHz': c_n0,
        'Eb_N0_dB': eb_n0
    }

# ============== BER CALCULATION ==============

def theoretical_cpfsk_ber_coherent(eb_n0_db, h=0.5):
    """
    Theoretical CPFSK BER (coherent detection)
    For h=0.5 (MSK), same as BPSK
    """
    from scipy.special import erfc
    eb_n0 = 10 ** (eb_n0_db / 10)
    # Coherent MSK BER = same as BPSK
    return 0.5 * erfc(np.sqrt(eb_n0))

def theoretical_cpfsk_ber_noncoherent(eb_n0_db, h=0.5):
    """
    Theoretical CPFSK BER (non-coherent detection)
    Approximation for discriminator detection
    """
    eb_n0 = 10 ** (eb_n0_db / 10)
    # Non-coherent FSK BER approximation
    return 0.5 * np.exp(-eb_n0 * h**2 / 2)

# ============== PHASE TRELLIS VISUALIZATION ==============

def generate_phase_trellis(bits, h=MOD_INDEX):
    """
    Generate phase trellis for CPFSK
    Shows how phase evolves with each symbol
    """
    phase = [0]
    for bit in bits:
        delta = h * np.pi if bit == 1 else -h * np.pi
        phase.append(phase[-1] + delta)

    return np.array(phase)

def meo_cpfsk_isl_simulation(num_bits=10000, distance_km=15000, doppler_hz=5000, use_rf_chain=True):
    """
    Complete MEO CPFSK ISL system simulation with RF front-end
    TX: Encoding -> Modulation -> Up-conversion to 2.2 GHz -> PA
    Channel: FSPL at carrier freq -> Doppler -> Drift -> AWGN
    RX: LNA -> Down-conversion to baseband -> Demodulation -> Decoding
    """
    print(f"MEO CPFSK ISL Simulation {'(Full RF Chain)' if use_rf_chain else '(Baseband)'}")
    print(f"Carrier: {CARRIER_FREQ/1e9:.1f} GHz | Distance: {distance_km} km | Doppler: {doppler_hz/1000:.1f} kHz")
    if use_rf_chain:
        print(f"Baseband: {BASEBAND_SAMPLE_RATE/1e6:.0f} MHz | RF: {RF_SAMPLE_RATE/1e6:.0f} MHz | IF: {IF_FREQ/1e6:.0f} MHz")

    # Link budget
    budget = calculate_link_budget(distance_km)
    print(f"Link: EIRP={budget['EIRP_dBW']:.1f} dBW | FSPL={budget['FSPL_dB']:.1f} dB | Eb/N0={budget['Eb_N0_dB']:.1f} dB")

    # Generate random data
    np.random.seed(42)  # For reproducibility
    tx_bits = np.random.randint(0, 2, num_bits)

    if use_rf_chain:
        # Transmitter with RF up-conversion
        tx_rf, tx_baseband, tx_coded, rect = meo_cpfsk_transmitter(tx_bits, include_rf_upconversion=True)
        num_coded_bits = len(tx_coded)

        # RF Channel at carrier frequency
        rx_rf = meo_rf_channel(tx_rf, distance_km, doppler_hz, budget['Eb_N0_dB'])

        # Receiver with RF down-conversion
        rx_bits, rx_symbols, rx_baseband = meo_cpfsk_receiver(rx_rf, num_coded_bits, num_bits, doppler_hz, is_rf_signal=True)
    else:
        # Baseband-only simulation (legacy)
        tx_baseband, tx_coded, rect = meo_cpfsk_transmitter(tx_bits, include_rf_upconversion=False)
        num_coded_bits = len(tx_coded)
        rx_baseband_ch = meo_space_channel(tx_baseband, distance_km, doppler_hz, budget['Eb_N0_dB'])
        rx_bits, rx_symbols, _ = meo_cpfsk_receiver(rx_baseband_ch, num_coded_bits, num_bits, doppler_hz, is_rf_signal=False)

    # Calculate BER
    min_len = min(len(tx_bits), len(rx_bits))
    bit_errors = np.sum(tx_bits[:min_len] != rx_bits[:min_len])
    ber = bit_errors / min_len

    print(f"Results: TX={len(tx_bits)} | Coded={num_coded_bits} | Errors={bit_errors} | BER={ber:.2e}")
    print(f"Theory: BER_coh={theoretical_cpfsk_ber_coherent(budget['Eb_N0_dB']):.2e} | BER_noncoh={theoretical_cpfsk_ber_noncoherent(budget['Eb_N0_dB']):.2e}")

    return {'tx_bits': tx_bits, 'rx_bits': rx_bits, 'ber': ber, 'link_budget': budget}

CPFSK Space Environment Advantages
  • Saturated Amplifier: No back-off required = 3-8 dB more EIRP
  • Power Efficiency: Class C amplifier operation possible
  • Simpler Design: No Gaussian filter - rectangular pulses
  • MSK Optimal: h=0.5 achieves orthogonality and BPSK BER
  • Proven Heritage: Deep space, GPS, military satcom
  • Coherent Option: Can use optimal MSK receiver for best BER
Key Differences: CPFSK vs GFSK
  • Pulse Shape: Rectangular (CPFSK) vs Gaussian (GFSK)
  • Spectral Width: CPFSK has wider spectrum (more sidelobes)
  • ISI: No ISI in CPFSK (L=1), GFSK has controlled ISI
  • Complexity: CPFSK simpler (no filter design param BT)
  • MSK Detection: CPFSK h=0.5 supports optimal coherent MSK RX
  • BER Performance: CPFSK (coh) = BPSK; GFSK ~1dB worse
