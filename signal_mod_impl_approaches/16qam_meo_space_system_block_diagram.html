# ============== RF SYSTEM PARAMETERS ==============
CARRIER_FREQ = 27e9             # 27 GHz Ka-band
WAVELENGTH = c / CARRIER_FREQ
SYMBOL_RATE = 10e6              # 10 MSym/s
BITS_PER_SYMBOL = 4             # 16-QAM: 4 bits per symbol
BIT_RATE = SYMBOL_RATE * BITS_PER_SYMBOL  # 40 Mbps
SAMPLES_PER_SYMBOL = 8
SAMPLE_RATE = SYMBOL_RATE * SAMPLES_PER_SYMBOL
ROLLOFF = 0.20                  # RRC roll-off factor (tighter for QAM)
FILTER_SPAN = 12                # symbols

# ============== UP/DOWN CONVERSION PARAMETERS ==============
# For practical simulation, we use an intermediate frequency (IF) representation
IF_FREQ = 100e6                 # 100 MHz intermediate frequency for simulation
IF_SAMPLES_PER_SYMBOL = 32      # Higher oversampling for IF signal
IF_SAMPLE_RATE = SYMBOL_RATE * IF_SAMPLES_PER_SYMBOL  # 320 MHz IF sample rate

# Link budget parameters
TX_POWER_W = 25                  # 25W SSPA

TX_ANTENNA_GAIN_DBI = 44        # 0.6m dish @ 27 GHz
RX_ANTENNA_GAIN_DBI = 44

LINE_LOSS_DB = 1.5

# 16-QAM constellation levels
QAM16_LEVELS = np.array([-3, -1, 1, 3])
QAM16_NORM = np.sqrt(10)  # Normalization factor for unit average power

# Gray-coded 16-QAM mapping
# Bits b3b2 map to I, bits b1b0 map to Q
GRAY_MAP = {0: -3, 1: -1, 3: 1, 2: 3}  # 00->-3, 01->-1, 11->+1, 10->+3
GRAY_DEMAP = {-3: 0, -1: 1, 1: 3, 3: 2}  # Inverse mapping

# ============== 16-QAM MODULATION ==============

def qam16_modulate(bits):
    """
    16-QAM modulation: map 4-bit groups to complex symbols
    Gray coding: adjacent symbols differ by 1 bit
    """
    # Pad to multiple of 4 bits
    pad_len = (4 - len(bits) % 4) % 4
    if pad_len > 0:
        bits = np.append(bits, np.zeros(pad_len, dtype=int))

    # Reshape into 4-bit groups
    bit_groups = bits.reshape(-1, 4)

    symbols = []
    for group in bit_groups:
        # Extract I bits (b3, b2) and Q bits (b1, b0)
        i_bits = group[0] * 2 + group[1]  # b3b2 as integer
        q_bits = group[2] * 2 + group[3]  # b1b0 as integer

        # Gray-coded mapping
        i_level = GRAY_MAP[i_bits]
        q_level = GRAY_MAP[q_bits]

        # Create normalized complex symbol
        symbol = (i_level + 1j * q_level) / QAM16_NORM
        symbols.append(symbol)

    return np.array(symbols)

def qam16_demodulate(symbols):
    """
    16-QAM demodulation: hard decision on I/Q levels
    Returns 4-bit groups based on nearest constellation point
    """
    bits = []
    thresholds = [-2, 0, 2]

    for s in symbols:
        # Denormalize
        s_denorm = s * QAM16_NORM

        # Slice I and Q components
        i_idx = np.digitize(np.real(s_denorm), thresholds)
        q_idx = np.digitize(np.imag(s_denorm), thresholds)

        i_level = QAM16_LEVELS[i_idx]
        q_level = QAM16_LEVELS[q_idx]

        # Gray demap to bits
        i_bits = GRAY_DEMAP[i_level]
        q_bits = GRAY_DEMAP[q_level]

        # Convert to individual bits
        bits.extend([(i_bits >> 1) & 1, i_bits & 1,
                     (q_bits >> 1) & 1, q_bits & 1])

    return np.array(bits)

def qam16_soft_demodulate(symbols, noise_var):
    """
    Soft 16-QAM demodulation: compute LLRs for LDPC decoder
    """
    llr = np.zeros(len(symbols) * 4)

    # Generate all 16 constellation points
    constellation = []
    bit_patterns = []
    for i in range(16):
        i_bits = (i >> 2) & 3
        q_bits = i & 3
        i_level = GRAY_MAP[i_bits]
        q_level = GRAY_MAP[q_bits]
        constellation.append((i_level + 1j * q_level) / QAM16_NORM)
        bit_patterns.append([(i >> 3) & 1, (i >> 2) & 1,
                             (i >> 1) & 1, i & 1])

    constellation = np.array(constellation)
    bit_patterns = np.array(bit_patterns)

    for i, s in enumerate(symbols):
        # Calculate distances to all constellation points
        distances = np.abs(s - constellation) ** 2

        # For each of the 4 bits, calculate LLR
        for b in range(4):
            mask_0 = bit_patterns[:, b] == 0
            mask_1 = bit_patterns[:, b] == 1

            min_d0 = np.min(distances[mask_0])
            min_d1 = np.min(distances[mask_1])

            llr[4*i + b] = (min_d1 - min_d0) / (2 * noise_var)

    return llr

# ============== LDPC ENCODER/DECODER (Simplified) ==============

def ldpc_encode(bits, code_rate=0.75):
    """Simplified LDPC encoder (adds parity bits)"""
    k = len(bits)
    n = int(k / code_rate)
    parity_len = n - k

    parity = np.zeros(parity_len, dtype=int)
    for i in range(parity_len):
        parity[i] = np.sum(bits[i::parity_len]) % 2

    return np.concatenate([bits, parity])

def ldpc_decode(bits, code_rate=0.75):
    """Simplified LDPC decoder (extracts data bits)"""
    k = int(len(bits) * code_rate)
    return bits[:k]

# ============== INTERLEAVER ==============

def block_interleave(bits, rows=64):
    """Block interleaver for burst error protection"""
    cols = int(np.ceil(len(bits) / rows))
    padded = np.zeros(rows * cols, dtype=int)
    padded[:len(bits)] = bits
    matrix = padded.reshape(rows, cols)
    return matrix.T.flatten()

def block_deinterleave(bits, rows=64):
    """Block de-interleaver"""
    cols = int(np.ceil(len(bits) / rows))
    matrix = bits[:rows*cols].reshape(cols, rows)
    return matrix.T.flatten()

# ============== SCRAMBLER ==============

def lfsr_sequence(length, seed=0x1ACFFC1D):
    """Generate LFSR pseudo-random sequence"""
    state = seed
    seq = np.zeros(length, dtype=int)
    for i in range(length):
        seq[i] = state & 1
        newbit = ((state >> 0) ^ (state >> 2) ^ (state >> 3) ^ (state >> 5)) & 1
        state = (state >> 1) | (newbit << 31)
    return seq

def scramble(bits):
    pn = lfsr_sequence(len(bits))
    return (bits ^ pn).astype(int)

def descramble(bits):
    return scramble(bits)

# ============== PULSE SHAPING ==============

def rrc_filter(num_taps, alpha, sps):
    """Root Raised Cosine filter design"""
    t = np.arange(-num_taps//2, num_taps//2 + 1) / sps
    h = np.zeros_like(t, dtype=float)

    for i, ti in enumerate(t):
        if ti == 0:
            h[i] = 1 + alpha * (4/np.pi - 1)
        elif abs(abs(ti) - 1/(4*alpha)) < 1e-10:
            h[i] = alpha/np.sqrt(2) * ((1+2/np.pi)*np.sin(np.pi/(4*alpha)) +
                                         (1-2/np.pi)*np.cos(np.pi/(4*alpha)))
        else:
            num = np.sin(np.pi*ti*(1-alpha)) + 4*alpha*ti*np.cos(np.pi*ti*(1+alpha))
            den = np.pi*ti*(1 - (4*alpha*ti)**2)
            h[i] = num / (den + 1e-10)

    return h / np.sqrt(np.sum(h**2))

# ============== UP-CONVERSION (TX) ==============

def upsample_for_if(baseband_signal, baseband_sps, if_sps):
    """Upsample baseband signal to IF sample rate for up-conversion"""
    upsample_factor = if_sps // baseband_sps
    upsampled = np.zeros(len(baseband_signal) * upsample_factor, dtype=complex)
    upsampled[::upsample_factor] = baseband_signal
    # Anti-imaging lowpass filter
    nyq = 0.5 * if_sps / baseband_sps
    cutoff = 0.4 / nyq
    b, a = butter(5, cutoff, btype='low')
    upsampled_i = filtfilt(b, a, np.real(upsampled)) * upsample_factor
    upsampled_q = filtfilt(b, a, np.imag(upsampled)) * upsample_factor
    return upsampled_i + 1j * upsampled_q

def upconvert_to_if(baseband_signal, if_freq, sample_rate):
    """Up-convert complex baseband signal to IF"""
    t = np.arange(len(baseband_signal)) / sample_rate
    carrier = np.exp(2j * np.pi * if_freq * t)
    if_signal = baseband_signal * carrier
    return if_signal, t

def tx_upconversion(baseband_signal):
    """Complete transmitter up-conversion chain"""
    upsampled = upsample_for_if(baseband_signal, SAMPLES_PER_SYMBOL, IF_SAMPLES_PER_SYMBOL)
    if_signal, t = upconvert_to_if(upsampled, IF_FREQ, IF_SAMPLE_RATE)
    return if_signal, t

# ============== DOWN-CONVERSION (RX) ==============

def downconvert_from_if(if_signal, if_freq, sample_rate, freq_offset=0):
    """Down-convert IF signal back to complex baseband"""
    t = np.arange(len(if_signal)) / sample_rate
    lo_freq = if_freq + freq_offset
    lo_signal = np.exp(-2j * np.pi * lo_freq * t)
    baseband_raw = if_signal * lo_signal
    return baseband_raw

def downsample_from_if(if_baseband, if_sps, baseband_sps):
    """Downsample from IF sample rate back to baseband sample rate"""
    downsample_factor = if_sps // baseband_sps
    nyq = 0.5
    cutoff = 0.4 / (if_sps / baseband_sps / 2)
    cutoff = min(cutoff, 0.99)
    b, a = butter(5, cutoff, btype='low')
    filtered_i = filtfilt(b, a, np.real(if_baseband))
    filtered_q = filtfilt(b, a, np.imag(if_baseband))
    downsampled = (filtered_i + 1j * filtered_q)[::downsample_factor]
    return downsampled

def rx_downconversion(if_signal, doppler_hz_baseband=0):
    """Complete receiver down-conversion chain with Doppler compensation"""
    doppler_if = doppler_hz_baseband * (IF_FREQ / CARRIER_FREQ)
    baseband_raw = downconvert_from_if(if_signal, IF_FREQ, IF_SAMPLE_RATE, doppler_if)
    baseband_signal = downsample_from_if(baseband_raw, IF_SAMPLES_PER_SYMBOL, SAMPLES_PER_SYMBOL)
    return baseband_signal

# ============== DIGITAL PREDISTORTION ==============

def sspa_model(signal_in, p_sat=1.0, smoothness=2):
    """
    Rapp SSPA model for AM/AM distortion
    smoothness parameter p controls sharpness of saturation
    """
    amplitude = np.abs(signal_in)
    phase = np.angle(signal_in)

    # Rapp model AM/AM
    gain = amplitude / ((1 + (amplitude / p_sat) ** (2 * smoothness)) ** (1 / (2 * smoothness)))

    return gain * np.exp(1j * phase)

def digital_predistort(signal_in, backoff_db=6):
    """
    Digital predistortion for SSPA linearization
    Applies inverse nonlinearity with back-off
    """
    backoff_linear = 10 ** (-backoff_db / 20)
    return signal_in * backoff_linear

# ============== SPACE CHANNEL MODEL ==============

def free_space_path_loss_db(distance, freq=CARRIER_FREQ):
    """Calculate FSPL in dB"""
    fspl = (4 * np.pi * distance * freq / c) ** 2
    return 10 * np.log10(fspl)

def apply_doppler_shift(signal_in, doppler_hz, sample_rate):
    """Apply Doppler frequency shift"""
    t = np.arange(len(signal_in)) / sample_rate
    shift = np.exp(2j * np.pi * doppler_hz * t)
    return signal_in * shift

def phase_noise(signal_len, phase_noise_std_rad=0.005):
    """
    Generate phase noise (random walk model)
    Note: 16-QAM is sensitive to phase noise
    """
    phase_increments = phase_noise_std_rad * np.random.randn(signal_len)
    phase = np.cumsum(phase_increments)
    return np.exp(1j * phase)

def iq_imbalance(signal_in, gain_imb_db=0.2, phase_imb_deg=0.5):
    """
    Apply I/Q gain and phase imbalance
    Tighter specs required for 16-QAM
    """
    g = 10 ** (gain_imb_db / 20)
    phi = np.deg2rad(phase_imb_deg)
    i = np.real(signal_in)
    q = np.imag(signal_in)
    i_out = i
    q_out = g * (np.sin(phi) * i + np.cos(phi) * q)
    return i_out + 1j * q_out

def meo_space_channel(signal_in, distance_km, doppler_hz, snr_db):
    """Complete MEO space channel model for 16-QAM (baseband version)"""
    sig = apply_doppler_shift(signal_in, doppler_hz, SAMPLE_RATE)
    sig = sig * phase_noise(len(sig))
    sig = iq_imbalance(sig, 0.15, 0.3)
    signal_power = np.mean(np.abs(sig) ** 2)
    snr_linear = 10 ** (snr_db / 10)
    noise_power = signal_power / snr_linear
    noise = np.sqrt(noise_power/2) * (np.random.randn(len(sig)) + 1j*np.random.randn(len(sig)))
    return sig + noise

def meo_rf_space_channel(if_signal, distance_km, doppler_hz_rf, snr_db):
    """MEO space channel model for up-converted RF/IF signal"""
    distance_m = distance_km * 1000
    sig = if_signal.copy()
    # Apply Doppler shift scaled to IF
    doppler_if = doppler_hz_rf * (IF_FREQ / CARRIER_FREQ)
    t = np.arange(len(sig)) / IF_SAMPLE_RATE
    doppler_shift = np.exp(2j * np.pi * doppler_if * t)
    sig = sig * doppler_shift
    # Phase noise (scaled for IF sample rate)
    phase_noise_std = 0.001 / np.sqrt(IF_SAMPLES_PER_SYMBOL)
    sig = sig * phase_noise(len(sig), phase_noise_std)
    # I/Q imbalance
    sig = iq_imbalance(sig, gain_imb_db=0.15, phase_imb_deg=0.3)
    # Scintillation (slow fading)
    scintillation_std = 0.02
    scintillation = 1 + scintillation_std * np.random.randn(len(sig))
    window_len = min(1000, len(sig) // 10)
    if window_len > 2:
        scintillation = np.convolve(scintillation, np.ones(window_len)/window_len, mode='same')
    sig = sig * scintillation
    # AWGN at IF
    signal_power = np.mean(np.abs(sig) ** 2)
    snr_linear = 10 ** (snr_db / 10)
    noise_power = signal_power / snr_linear
    noise = np.sqrt(noise_power/2) * (np.random.randn(len(sig)) + 1j*np.random.randn(len(sig)))
    return sig + noise

# ============== CARRIER RECOVERY ==============

def dd_carrier_pll(signal_in, loop_bw=0.005):
    """
    Decision-directed PLL for 16-QAM carrier recovery
    Uses detected symbols for phase error estimation
    """
    phase = 0
    freq = 0
    output = np.zeros_like(signal_in, dtype=complex)

    # Loop filter coefficients
    damping = 0.707
    K1 = 4 * damping * loop_bw
    K2 = 4 * loop_bw ** 2

    thresholds = np.array([-2, 0, 2]) / QAM16_NORM

    for i in range(len(signal_in)):
        # Mix with NCO
        output[i] = signal_in[i] * np.exp(-1j * phase)

        # Decision-directed: find nearest constellation point
        i_idx = np.digitize(np.real(output[i]), thresholds)
        q_idx = np.digitize(np.imag(output[i]), thresholds)

        i_dec = QAM16_LEVELS[i_idx] / QAM16_NORM
        q_dec = QAM16_LEVELS[q_idx] / QAM16_NORM
        dec_symbol = i_dec + 1j * q_dec

        # Phase error = Im(received * conj(decision))
        error = np.imag(output[i] * np.conj(dec_symbol))

        # Loop filter
        freq += K2 * error
        phase += K1 * error + freq

    return output

# ============== TRANSMITTER ==============

def meo_16qam_transmitter(data_bits, enable_upconversion=True):
    """Complete MEO 16-QAM satellite transmitter chain with IF up-conversion"""
    # LDPC encoding (rate 3/4)
    encoded = ldpc_encode(data_bits, code_rate=0.75)
    # Interleaving
    interleaved = block_interleave(encoded)
    # Scrambling
    scrambled = scramble(interleaved)
    # 16-QAM modulation
    symbols = qam16_modulate(scrambled)
    # Upsampling
    upsampled = np.zeros(len(symbols) * SAMPLES_PER_SYMBOL, dtype=complex)
    upsampled[::SAMPLES_PER_SYMBOL] = symbols
    # Pulse shaping (RRC filter)
    num_taps = FILTER_SPAN * SAMPLES_PER_SYMBOL + 1
    rrc = rrc_filter(num_taps, ROLLOFF, SAMPLES_PER_SYMBOL)
    baseband_shaped = np.convolve(upsampled, rrc, mode='same')
    # Digital predistortion for SSPA
    baseband_predistorted = digital_predistort(baseband_shaped, OUTPUT_BACKOFF_DB)

    if enable_upconversion:
        # TX UP-CONVERSION TO 27 GHz CARRIER (IF REPRESENTATION)
        if_signal, if_time = tx_upconversion(baseband_predistorted)
        # Apply SSPA model after up-conversion
        tx_signal = sspa_model(if_signal, p_sat=1.0, smoothness=2)
        return tx_signal, symbols, rrc, baseband_predistorted
    else:
        return baseband_predistorted, symbols, rrc, baseband_predistorted

# ============== RECEIVER ==============

def meo_16qam_receiver(rx_signal, rrc, num_symbols, num_info_bits, doppler_hz=0, enable_downconversion=True):
    """Complete MEO 16-QAM satellite receiver chain with IF down-conversion"""

    if enable_downconversion:
        # RX DOWN-CONVERSION FROM IF TO BASEBAND
        rx_baseband = rx_downconversion(rx_signal, doppler_hz)
    else:
        rx_baseband = rx_signal
        if doppler_hz != 0:
            rx_baseband = apply_doppler_shift(rx_baseband, -doppler_hz, SAMPLE_RATE)

    # Matched filter (RRC)
    filtered = np.convolve(rx_baseband, rrc, mode='same')
    # Downsample to symbol rate
    rx_symbols_raw = filtered[::SAMPLES_PER_SYMBOL][:num_symbols]
    # AGC normalization (before PLL)
    rx_symbols_raw = rx_symbols_raw / np.sqrt(np.mean(np.abs(rx_symbols_raw)**2))
    # Carrier recovery (DD-PLL on symbol-rate samples)
    rx_symbols = dd_carrier_pll(rx_symbols_raw)
    # 16-QAM demodulation (hard decision)
    rx_bits = qam16_demodulate(rx_symbols)
    # Descramble
    descrambled = descramble(rx_bits)
    # De-interleave
    deinterleaved = block_deinterleave(descrambled)
    # LDPC decode
    decoded = ldpc_decode(deinterleaved, code_rate=0.75)

    return decoded[:num_info_bits], rx_symbols, rx_baseband

# ============== LINK BUDGET ==============

def calculate_link_budget(distance_km):
    """Calculate link budget for 16-QAM MEO ISL"""
    distance = distance_km * 1000

    tx_power_dbw = 10 * np.log10(TX_POWER_W)
    tx_power_eff = tx_power_dbw - OUTPUT_BACKOFF_DB  # Account for OBO
    eirp = tx_power_eff + TX_ANTENNA_GAIN_DBI - LINE_LOSS_DB
    fspl = free_space_path_loss_db(distance)
    rx_gain = RX_ANTENNA_GAIN_DBI - LINE_LOSS_DB
    g_over_t = rx_gain - 10 * np.log10(SYSTEM_NOISE_TEMP_K)
    c_n0 = eirp - fspl - 0.5 + g_over_t + 228.6
    eb_n0 = c_n0 - 10 * np.log10(BIT_RATE)

    return {
        'EIRP_dBW': eirp,
        'FSPL_dB': fspl,
        'C_N0_dBHz': c_n0,
        'Eb_N0_dB': eb_n0
    }

# ============== BER CALCULATION ==============

def theoretical_16qam_ber(eb_n0_db):
    """
    Theoretical 16-QAM BER in AWGN
    """
    from scipy.special import erfc
    eb_n0 = 10 ** (eb_n0_db / 10)
    # For 16-QAM: Es = 4*Eb (4 bits per symbol)
    es_n0 = 4 * eb_n0
    # Approximate BER for 16-QAM
    return (3/8) * erfc(np.sqrt(es_n0 / 10))

def meo_16qam_isl_simulation(num_bits=40000, distance_km=10000, doppler_hz=60000, use_rf_channel=True):
    """Complete MEO 16-QAM ISL system simulation with RF up/down conversion"""

    print(f"MEO 16-QAM ISL Simulation")
    print(f"=========================")
    print(f"Mode: {'RF Up/Down Conversion' if use_rf_channel else 'Baseband'}")
    print(f"Carrier Frequency: {CARRIER_FREQ/1e9:.1f} GHz")
    if use_rf_channel:
        print(f"Simulation IF: {IF_FREQ/1e6:.1f} MHz")
        print(f"IF Sample Rate: {IF_SAMPLE_RATE/1e6:.1f} MHz")
    print(f"Distance: {distance_km} km")
    print(f"Doppler (at carrier): {doppler_hz/1000:.1f} kHz")
    print()

    # Link budget (calculated at 27 GHz carrier frequency)
    budget = calculate_link_budget(distance_km)
    print(f"Link Budget (at {CARRIER_FREQ/1e9:.0f} GHz):")
    print(f"  EIRP: {budget['EIRP_dBW']:.1f} dBW")
    print(f"  FSPL: {budget['FSPL_dB']:.1f} dB")
    print(f"  C/N0: {budget['C_N0_dBHz']:.1f} dBHz")
    print(f"  Eb/N0: {budget['Eb_N0_dB']:.1f} dB")
    print()

    # Generate random data
    num_info_bits = int(num_bits * 0.75)
    num_info_bits = (num_info_bits // 4) * 4
    tx_bits = np.random.randint(0, 2, num_info_bits)

    # Transmitter (with or without up-conversion)
    tx_signal, tx_symbols, rrc, tx_baseband = meo_16qam_transmitter(tx_bits, enable_upconversion=use_rf_channel)

    if use_rf_channel:
        # RF Channel (operates on up-converted signal)
        rx_signal = meo_rf_space_channel(tx_signal, distance_km, doppler_hz, budget['Eb_N0_dB'])
        # Receiver (with down-conversion)
        num_symbols = len(tx_symbols)
        rx_bits, rx_symbols, rx_baseband = meo_16qam_receiver(
            rx_signal, rrc, num_symbols, num_info_bits, doppler_hz, enable_downconversion=True)
    else:
        # Legacy baseband mode
        rx_signal = meo_space_channel(tx_signal, distance_km, doppler_hz, budget['Eb_N0_dB'])
        num_symbols = len(tx_symbols)
        rx_bits, rx_symbols, rx_baseband = meo_16qam_receiver(
            rx_signal, rrc, num_symbols, num_info_bits, doppler_hz, enable_downconversion=False)

    # Calculate BER
    min_len = min(len(tx_bits), len(rx_bits))
    bit_errors = np.sum(tx_bits[:min_len] != rx_bits[:min_len])
    ber = bit_errors / min_len

    print(f"Results:")
    print(f"  TX info bits: {len(tx_bits)}")
    print(f"  TX symbols: {len(tx_symbols)}")
    print(f"  RX bits: {len(rx_bits)}")
    print(f"  Bit errors: {bit_errors}")
    print(f"  BER: {ber:.2e}")
    print(f"  Theoretical BER (uncoded): {theoretical_16qam_ber(budget['Eb_N0_dB']):.2e}")

    return {
        'tx_bits': tx_bits, 'rx_bits': rx_bits,
        'tx_symbols': tx_symbols, 'rx_symbols': rx_symbols,
        'tx_baseband': tx_baseband, 'rx_baseband': rx_baseband,
        'tx_if_signal': tx_signal if use_rf_channel else None,
        'rx_if_signal': rx_signal if use_rf_channel else None,
        'ber': ber, 'link_budget': budget
    }

RF Up/Down Conversion Architecture
  • IF Simulation: 100 MHz IF represents 27 GHz carrier
  • TX Upsample: 80 MHz -> 320 MHz for IF processing
  • RX Downsample: 320 MHz -> 80 MHz after IF downconvert
  • Doppler Compensation: Applied at IF in LO frequency
  • Anti-Alias/Image: 5th order Butterworth filters
  • SSPA at IF: Rapp model after up-conversion
RF Channel Impairments Modeled
  • Doppler Shift: Scaled from RF to IF (IF_FREQ/CARRIER_FREQ)
  • Phase Noise: OCXO model, 0.001 rad/symbol period
  • I/Q Imbalance: 0.15 dB gain, 0.3 deg phase error
  • Scintillation: 2% amplitude variation (slow fading)
  • AWGN: Thermal noise at IF based on link budget
  • FSPL: Calculated at 27 GHz for link budget
