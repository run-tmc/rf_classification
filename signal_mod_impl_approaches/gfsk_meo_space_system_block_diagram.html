GFSK Mathematical Definition
Transmitted Signal:
s(t) = A * cos(2*pi*f_c*t + phi(t))
Instantaneous Phase:
phi(t) = 2*pi*h * integral(g(tau)*d(tau) dtau)
Gaussian Pulse Shape:
g(t) = (sqrt(2*pi)/T) * (B/sqrt(ln(2))) * exp(-2*(pi*B*t)^2/ln(2))
Where:
h = modulation index (0.5 for MSK)
B = 3dB bandwidth of Gaussian filter
T = symbol period
BT = bandwidth-time product
Frequency Deviation:
delta_f = h * R_b / 2 = h / (2*T)

# ============== GAUSSIAN FILTER DESIGN ==============

def gaussian_filter(bt_product, sps, num_symbols=4):
    """
    Design Gaussian filter for GFSK modulation
    bt_product: Bandwidth-Time product (typically 0.3-0.5)
    sps: samples per symbol
    num_symbols: filter span in symbols
    """
    # Time vector (centered at 0)
    num_taps = num_symbols * sps + 1
    t = np.arange(-(num_taps//2), num_taps//2 + 1) / sps

    # Gaussian filter impulse response
    alpha = np.sqrt(np.log(2) / 2) / (bt_product)
    h = np.sqrt(np.pi) / alpha * np.exp(-(np.pi * t / alpha) ** 2)

    # Normalize to unit area (frequency pulse)
    h = h / np.sum(h)

    return h

def gaussian_frequency_pulse(bt_product, sps, num_symbols=4):
    """
    Generate Gaussian frequency pulse shape
    This is the derivative of the phase pulse
    """
    num_taps = num_symbols * sps + 1
    t = np.arange(-(num_taps//2), num_taps//2 + 1) / sps

    # Gaussian frequency pulse
    sigma = np.sqrt(np.log(2)) / (2 * np.pi * bt_product)
    h = (1 / (np.sqrt(2 * np.pi) * sigma)) * np.exp(-t**2 / (2 * sigma**2))

    # Normalize
    h = h / np.sum(h)

    return h

# ============== GFSK MODULATION ==============

def gfsk_modulate(bits, sps=SAMPLES_PER_SYMBOL, h=MOD_INDEX, bt=BT_PRODUCT):
    """
    GFSK modulation
    bits: input binary data (0/1)
    sps: samples per symbol
    h: modulation index
    bt: bandwidth-time product

    Returns: complex baseband GFSK signal (constant envelope)
    """
    # Convert bits to bipolar NRZ (+1/-1)
    nrz = 2 * bits.astype(float) - 1

    # Upsample
    upsampled = np.zeros(len(nrz) * sps)
    upsampled[::sps] = nrz

    # Apply Gaussian filter to get frequency pulse
    gauss = gaussian_frequency_pulse(bt, sps)
    freq_pulse = np.convolve(upsampled, gauss, mode='same')

    # Scale by modulation index and pi
    freq_scaled = freq_pulse * h * np.pi / sps

    # Integrate to get phase
    phase = np.cumsum(freq_scaled)

    # Generate complex baseband signal
    signal_out = np.exp(1j * phase)

    return signal_out, phase, gauss

def gfsk_demodulate(signal_in, sps=SAMPLES_PER_SYMBOL, bt=BT_PRODUCT):
    """
    GFSK demodulation using delay-and-multiply FM discriminator
    Non-coherent detection (robust for space applications)
    """
    # Delay-and-multiply FM discriminator (more robust than diff)
    delay = 1
    delayed = np.roll(signal_in, delay)
    product = signal_in * np.conj(delayed)
    freq = np.angle(product)

    # Apply matched Gaussian filter
    gauss = gaussian_frequency_pulse(bt, sps)
    filtered = np.convolve(freq, gauss, mode='same')

    # Symbol timing recovery: eye opening search
    num_test = min(200, len(filtered) // sps - 1)
    best_phase = 0
    max_eye = 0
    for phase in range(sps):
        samples = filtered[phase:phase + num_test * sps:sps]
        if len(samples) > 10:
            pos = samples[samples > 0]
            neg = samples[samples <= 0]
            if len(pos) > 5 and len(neg) > 5:
                eye = np.mean(pos) - np.mean(neg)
                if eye > max_eye:
                    max_eye, best_phase = eye, phase

    symbols = filtered[best_phase::sps]
    bits = (symbols > 0).astype(int)
    return bits, symbols

# ============== FM DISCRIMINATOR (Alternative) ==============

def fm_discriminator_delay_multiply(signal_in, delay_samples=1):
    """
    FM discriminator using delay and multiply method
    More robust to noise than direct differentiation
    """
    # Delay signal
    delayed = np.roll(signal_in, delay_samples)

    # Multiply conjugate
    product = signal_in * np.conj(delayed)

    # Extract phase (imaginary part of log, or atan2)
    freq_out = np.angle(product)

    return freq_out

# ============== CONVOLUTIONAL ENCODER ==============

def conv_encode(bits, g1=0o171, g2=0o133):
    """
    Convolutional encoder - NASA standard rate 1/2, K=7
    """
    K = 7
    state = 0
    encoded = []

    for bit in bits:
        state = ((state << 1) | bit) & (2**K - 1)
        out1 = bin(state & g1).count('1') % 2
        out2 = bin(state & g2).count('1') % 2
        encoded.extend([out1, out2])

    # Flush encoder
    for _ in range(K - 1):
        state = (state << 1) & (2**K - 1)
        out1 = bin(state & g1).count('1') % 2
        out2 = bin(state & g2).count('1') % 2
        encoded.extend([out1, out2])

    return np.array(encoded)

def viterbi_decode_hard(received, num_info_bits, g1=0o171, g2=0o133):
    """Hard-decision Viterbi decoder for rate 1/2, K=7 code"""
    K, num_states = 7, 64  # 2^(K-1) states
    if len(received) % 2: received = np.append(received, 0)
    num_sym = len(received) // 2

    path_metric = np.full(num_states, 1e9)
    path_metric[0] = 0
    survivors = np.zeros((num_sym, num_states, 2), dtype=int)

    for t in range(num_sym):
        rx = (int(received[2*t]), int(received[2*t + 1]))
        new_metric = np.full(num_states, 1e9)
        for state in range(num_states):
            if path_metric[state] >= 1e9: continue
            for bit in [0, 1]:
                enc = ((state << 1) | bit) & (2**K - 1)
                o1 = bin(enc & g1).count('1') % 2
                o2 = bin(enc & g2).count('1') % 2
                next_st = enc & (num_states - 1)
                bm = (o1 != rx[0]) + (o2 != rx[1])
                cand = path_metric[state] + bm
                if cand < new_metric[next_st]:
                    new_metric[next_st] = cand
                    survivors[t, next_st] = [state, bit]
        path_metric = new_metric

    # Traceback
    cur = 0 if path_metric[0] < 1e9 else np.argmin(path_metric)
    decoded = np.zeros(num_sym, dtype=int)
    for t in range(num_sym - 1, -1, -1):
        decoded[t] = survivors[t, cur, 1]
        cur = survivors[t, cur, 0]
    return decoded[:num_info_bits]

# ============== INTERLEAVER ==============

def block_interleave(bits, rows=32):
    """Block interleaver for burst error protection"""
    cols = int(np.ceil(len(bits) / rows))
    padded = np.zeros(rows * cols, dtype=int)
    padded[:len(bits)] = bits
    matrix = padded.reshape(rows, cols)
    return matrix.T.flatten()

def block_deinterleave(bits, rows=32):
    """Block de-interleaver"""
    cols = int(np.ceil(len(bits) / rows))
    matrix = bits[:rows*cols].reshape(cols, rows)
    return matrix.T.flatten()

# ============== SCRAMBLER ==============

def lfsr_sequence(length, seed=0x1ACFFC1D):
    """Generate LFSR pseudo-random sequence"""
    state = seed
    seq = np.zeros(length, dtype=int)
    for i in range(length):
        seq[i] = state & 1
        newbit = ((state >> 0) ^ (state >> 2) ^ (state >> 3) ^ (state >> 5)) & 1
        state = (state >> 1) | (newbit << 31)
    return seq

def scramble(bits):
    pn = lfsr_sequence(len(bits))
    return (bits ^ pn).astype(int)

def descramble(bits):
    return scramble(bits)

# ============== SPACE CHANNEL MODEL ==============

def free_space_path_loss_db(distance, freq=CARRIER_FREQ):
    """Calculate FSPL in dB"""
    fspl = (4 * np.pi * distance * freq / c) ** 2
    return 10 * np.log10(fspl)

def apply_doppler_shift(signal_in, doppler_hz, sample_rate):
    """Apply Doppler frequency shift"""
    t = np.arange(len(signal_in)) / sample_rate
    shift = np.exp(2j * np.pi * doppler_hz * t)
    return signal_in * shift

def apply_frequency_drift(signal_in, drift_rate_hz_per_s, sample_rate):
    """Apply slow frequency drift (oscillator aging/thermal)"""
    t = np.arange(len(signal_in)) / sample_rate
    phase = 2 * np.pi * drift_rate_hz_per_s * t**2 / 2
    return signal_in * np.exp(1j * phase)

def meo_space_channel(signal_in, distance_km, doppler_hz, snr_db):
    """Complete MEO space channel model for GFSK"""
    # Apply Doppler shift
    sig = apply_doppler_shift(signal_in, doppler_hz, SAMPLE_RATE)

    # Apply small frequency drift
    sig = apply_frequency_drift(sig, 10, SAMPLE_RATE)  # 10 Hz/s drift

    # Add AWGN
    signal_power = np.mean(np.abs(sig) ** 2)
    snr_linear = 10 ** (snr_db / 10)
    noise_power = signal_power / snr_linear
    noise = np.sqrt(noise_power/2) * (np.random.randn(len(sig)) + 1j*np.random.randn(len(sig)))

    return sig + noise

# ============== AFC (Automatic Frequency Control) ==============

def afc_estimate(signal_in, sps, sample_rate=BASEBAND_SAMPLE_RATE):
    """
    Estimate frequency offset using power-of-4 method
    For GFSK h=0.5, signal^4 removes modulation
    """
    # Raise to 4th power to remove modulation
    sig_power = signal_in ** 4
    n_fft = min(len(sig_power), 8192)
    spectrum = np.fft.fftshift(np.fft.fft(sig_power[:n_fft]))
    freqs = np.fft.fftshift(np.fft.fftfreq(n_fft, 1/sample_rate))

    # Find peak and divide by 4
    peak_idx = np.argmax(np.abs(spectrum))
    freq_offset = freqs[peak_idx] / 4

    # Limit to reasonable range
    max_offset = sample_rate / 4
    if abs(freq_offset) > max_offset:
        freq_offset = 0
    return freq_offset

# ============== TRANSMITTER ==============

def meo_gfsk_transmitter(data_bits, include_upconversion=True):
    """Complete MEO GFSK satellite transmitter with RF upconversion"""
    # Convolutional encoding (rate 1/2)
    encoded = conv_encode(data_bits)
    # Interleaving
    interleaved = block_interleave(encoded)
    # Scrambling
    scrambled = scramble(interleaved)
    # GFSK modulation
    baseband, phase, gauss = gfsk_modulate(scrambled)

    if not include_upconversion:
        return baseband, scrambled, gauss

    # RF Upconversion to carrier
    rf_signal, rf_time, i_up, q_up = upconvert_to_carrier(
        baseband, CARRIER_FREQ, BASEBAND_SAMPLE_RATE, RF_SAMPLE_RATE)
    tx_info = {'baseband': baseband, 'coded_bits': scrambled}
    return rf_signal, rf_time, tx_info

# ============== RECEIVER ==============

def meo_gfsk_receiver_rf(rf_signal, num_coded_bits, num_info_bits,
                          doppler_hz=0, is_real=True):
    """Complete MEO GFSK receiver with RF downconversion"""
    # RF Downconversion with Doppler pre-compensation
    lo_offset = -doppler_hz if doppler_hz != 0 else 0
    baseband, i_bb, q_bb = downconvert_to_baseband(
        rf_signal, CARRIER_FREQ, RF_SAMPLE_RATE,
        BASEBAND_SAMPLE_RATE, lo_offset)

    # AFC fine frequency tracking (power-of-4 method)
    freq_residual = afc_estimate(baseband, SAMPLES_PER_SYMBOL)
    if abs(freq_residual) > 100:
        baseband = apply_doppler_shift(baseband, -freq_residual,
                                        BASEBAND_SAMPLE_RATE)

    # GFSK demodulation (delay-multiply FM discriminator)
    rx_bits, symbols = gfsk_demodulate(baseband)
    rx_bits = rx_bits[:num_coded_bits]

    # Descramble, De-interleave, Viterbi decode
    descrambled = descramble(rx_bits)
    deinterleaved = block_deinterleave(descrambled)
    decoded = viterbi_decode_hard(deinterleaved, num_info_bits)

    return decoded[:num_info_bits], {'symbols': symbols}

# ============== LINK BUDGET ==============

def calculate_link_budget(distance_km):
    """Calculate link budget for GFSK MEO ISL"""
    distance = distance_km * 1000

    tx_power_dbw = 10 * np.log10(TX_POWER_W)
    # No back-off needed for constant envelope!
    eirp = tx_power_dbw + TX_ANTENNA_GAIN_DBI - LINE_LOSS_DB
    fspl = free_space_path_loss_db(distance)
    rx_gain = RX_ANTENNA_GAIN_DBI - LINE_LOSS_DB
    g_over_t = rx_gain - 10 * np.log10(SYSTEM_NOISE_TEMP_K)
    c_n0 = eirp - fspl - 0.5 + g_over_t + 228.6  # Boltzmann
    eb_n0 = c_n0 - 10 * np.log10(SYMBOL_RATE)

    return {
        'EIRP_dBW': eirp,
        'FSPL_dB': fspl,
        'C_N0_dBHz': c_n0,
        'Eb_N0_dB': eb_n0
    }

# ============== BER CALCULATION ==============

def theoretical_gfsk_ber(eb_n0_db, h=0.5):
    """
    Theoretical GFSK BER (non-coherent detection)
    Approximation for h=0.5
    """
    from scipy.special import erfc
    eb_n0 = 10 ** (eb_n0_db / 10)
    # Non-coherent GFSK BER approximation
    return 0.5 * np.exp(-0.5 * eb_n0 * h**2)

        rf_tx, rf_time, tx_info = meo_gfsk_transmitter(tx_bits, True)
        num_coded = len(tx_info['coded_bits'])
        rf_rx, ch_info = meo_rf_space_channel(rf_tx, distance_km,
                                               doppler_hz, budget['Eb_N0_dB'])
        rx_bits, rx_info = meo_gfsk_receiver_rf(rf_rx, num_coded, num_bits,
                                                 doppler_hz, is_real=True)
    else:
        # Baseband-only simulation
        tx_sig, tx_coded, gauss = meo_gfsk_transmitter(tx_bits, False)
        num_coded = len(tx_coded)
        rx_sig = meo_space_channel(tx_sig, distance_km, doppler_hz,
                                    budget['Eb_N0_dB'])
        rx_bits, _ = meo_gfsk_receiver(rx_sig, num_coded, num_bits, doppler_hz)

    # Calculate BER
    min_len = min(len(tx_bits), len(rx_bits))
    errors = np.sum(tx_bits[:min_len] != rx_bits[:min_len])
    ber = errors / min_len
    print(f"BER: {ber:.2e} ({errors}/{min_len} errors)")

    return {'ber': ber, 'tx_bits': tx_bits, 'rx_bits': rx_bits}

GFSK Space Environment Advantages
  • Saturated Amplifier: No back-off required = 3-8 dB more EIRP
  • Power Efficiency: Class C amplifier operation possible
  • Simple Demodulation: Non-coherent FM discriminator
  • Doppler Tolerant: AFC handles large frequency offsets
  • Proven Heritage: Bluetooth, GPS L1C, IoT satellites
  • Spectral Efficiency: Gaussian filter reduces OOB emissions
Key Implementation Details
  • FM Discriminator: Delay-and-multiply (z[n]*conj(z[n-1]))
  • AFC Method: Power-of-4 removes GFSK modulation
  • Timing Recovery: Eye-opening search for optimal phase
  • Viterbi Decoder: 64-state trellis with traceback
  • RF Chain: 10x oversample (800 MSa/s), 100 MHz IF
  • Doppler Range: Handles up to ±50 kHz offset
