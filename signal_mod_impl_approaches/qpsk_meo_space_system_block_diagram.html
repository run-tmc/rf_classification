    (1, 0): (-1 + 1j) / np.sqrt(2),  # 225 deg

# ============== QPSK MODULATION ==============

def bits_to_symbols(bits):
    """Convert bit stream to dibit pairs"""
    # Ensure even number of bits
    if len(bits) % 2 != 0:
        bits = np.append(bits, 0)
    return bits.reshape(-1, 2)

def qpsk_modulate(bits):
    """
    QPSK modulation with Gray coding
    Maps dibits to complex symbols
    00 -> (-1-j)/sqrt2, 01 -> (+1-j)/sqrt2
    10 -> (-1+j)/sqrt2, 11 -> (+1+j)/sqrt2
    """
    dibits = bits_to_symbols(bits)
    symbols = np.zeros(len(dibits), dtype=complex)

    for i, (b0, b1) in enumerate(dibits):
        symbols[i] = QPSK_CONSTELLATION[(b0, b1)]

    return symbols

def qpsk_modulate_fast(bits):
    """Vectorized QPSK modulation for performance"""
    if len(bits) % 2 != 0:
        bits = np.append(bits, 0)

    # Map bits to I and Q components
    I_bits = bits[0::2]  # Even bits -> Q
    Q_bits = bits[1::2]  # Odd bits -> I

    # Convert to bipolar: 0 -> -1, 1 -> +1
    I = (2 * Q_bits - 1) / np.sqrt(2)
    Q = -(2 * I_bits - 1) / np.sqrt(2)

    return I + 1j * Q

def qpsk_demodulate(symbols):
    """
    QPSK demodulation: hard decision on I/Q
    Returns bit stream from complex symbols
    """
    # Decision based on sign of I and Q
    I = np.real(symbols)
    Q = np.imag(symbols)

    # Map back to bits (Gray coded)
    Q_bits = (Q < 0).astype(int)   # First bit of dibit
    I_bits = (I > 0).astype(int)   # Second bit of dibit

    # Interleave to form bit stream
    bits = np.zeros(2 * len(symbols), dtype=int)
    bits[0::2] = Q_bits
    bits[1::2] = I_bits

    return bits

# ============== DIFFERENTIAL ENCODING (QPSK) ==============

def differential_encode_qpsk(symbols):
    """
    Differential encoding for QPSK to resolve 90 degree phase ambiguity
    Encodes the phase difference between consecutive symbols
    """
    encoded = np.zeros(len(symbols) + 1, dtype=complex)
    encoded[0] = 1 + 0j  # Reference symbol

    for i in range(len(symbols)):
        encoded[i+1] = encoded[i] * symbols[i]
        # Normalize to unit circle
        encoded[i+1] /= np.abs(encoded[i+1])

    return encoded[1:]

def differential_decode_qpsk(symbols):
    """Differential decoding for QPSK"""
    decoded = np.zeros(len(symbols) - 1, dtype=complex)

    for i in range(len(symbols) - 1):
        decoded[i] = symbols[i+1] * np.conj(symbols[i])
        # Normalize
        decoded[i] /= np.abs(decoded[i]) if np.abs(decoded[i]) > 0 else 1

    return decoded

# ============== CONVOLUTIONAL ENCODER (Rate 1/2, K=7) ==============

def conv_encode(bits, g1=0o171, g2=0o133):
    """
    Convolutional encoder - NASA standard rate 1/2, K=7
    g1 = 171 octal (1111001)
    g2 = 133 octal (1011011)
    """
    K = 7  # Constraint length
    state = 0
    encoded = []

    for bit in bits:
        state = ((state << 1) | int(bit)) & (2**K - 1)

        # Output 1: XOR of bits selected by g1
        out1 = bin(state & g1).count('1') % 2
        # Output 2: XOR of bits selected by g2
        out2 = bin(state & g2).count('1') % 2

        encoded.extend([out1, out2])

    # Flush encoder with zeros
    for _ in range(K - 1):
        state = (state << 1) & (2**K - 1)
        out1 = bin(state & g1).count('1') % 2
        out2 = bin(state & g2).count('1') % 2
        encoded.extend([out1, out2])

    return np.array(encoded)

def viterbi_decode(received, num_info_bits, g1=0o171, g2=0o133):
    """
    Viterbi decoder for rate 1/2, K=7 convolutional code
    Hard decision decoding
    """
    K = 7
    num_states = 2 ** (K - 1)

    # Initialize path metrics
    path_metric = np.full(num_states, np.inf)
    path_metric[0] = 0

    # Survivor paths
    survivors = np.zeros((len(received)//2, num_states), dtype=int)

    # Process received bits in pairs
    for i in range(0, len(received), 2):
        rx_bits = received[i:i+2]
        new_path_metric = np.full(num_states, np.inf)

        for state in range(num_states):
            for input_bit in [0, 1]:
                # Calculate next state
                next_state = ((state << 1) | input_bit) & (num_states - 1)

                # Calculate expected output
                full_state = (state << 1) | input_bit
                exp_out1 = bin(full_state & g1).count('1') % 2
                exp_out2 = bin(full_state & g2).count('1') % 2

                # Hamming distance
                branch_metric = (rx_bits[0] != exp_out1) + (rx_bits[1] != exp_out2)

                # Update path metric
                candidate_metric = path_metric[state] + branch_metric
                if candidate_metric < new_path_metric[next_state]:
                    new_path_metric[next_state] = candidate_metric
                    survivors[i//2, next_state] = state

        path_metric = new_path_metric

    # Traceback
    state = 0  # Assume terminated to zero state
    decoded = []
    for i in range(len(received)//2 - 1, -1, -1):
        prev_state = survivors[i, state]
        decoded_bit = (state >> (K - 2)) & 1
        decoded.append(decoded_bit)
        state = prev_state

    decoded = np.array(decoded[::-1])
    return decoded[:num_info_bits]

# ============== INTERLEAVER ==============

def block_interleave(bits, rows=32):
    """Block interleaver for burst error protection"""
    cols = int(np.ceil(len(bits) / rows))
    padded = np.zeros(rows * cols, dtype=int)
    padded[:len(bits)] = bits
    matrix = padded.reshape(rows, cols)
    return matrix.T.flatten()

def block_deinterleave(bits, rows=32):
    """Block de-interleaver"""
    cols = int(np.ceil(len(bits) / rows))
    matrix = bits[:rows*cols].reshape(cols, rows)
    return matrix.T.flatten()

# ============== SCRAMBLER ==============

def lfsr_sequence(length, seed=0x1ACFFC1D):
    """Generate LFSR pseudo-random sequence"""
    state = seed
    seq = np.zeros(length, dtype=int)
    for i in range(length):
        seq[i] = state & 1
        newbit = ((state >> 0) ^ (state >> 2) ^ (state >> 3) ^ (state >> 5)) & 1
        state = (state >> 1) | (newbit << 31)
    return seq

def scramble(bits):
    """Scramble bits with LFSR sequence"""
    pn = lfsr_sequence(len(bits))
    return (bits ^ pn).astype(int)

def descramble(bits):
    """Descramble bits"""
    return scramble(bits)

# ============== PULSE SHAPING ==============

def rrc_filter(num_taps, alpha, sps):
    """Root Raised Cosine filter design"""
    t = np.arange(-num_taps//2, num_taps//2 + 1) / sps
    h = np.zeros_like(t, dtype=float)

    for i, ti in enumerate(t):
        if ti == 0:
            h[i] = 1 + alpha * (4/np.pi - 1)
        elif abs(abs(ti) - 1/(4*alpha)) < 1e-10:
            h[i] = alpha/np.sqrt(2) * ((1+2/np.pi)*np.sin(np.pi/(4*alpha)) +
                                         (1-2/np.pi)*np.cos(np.pi/(4*alpha)))
        else:
            num = np.sin(np.pi*ti*(1-alpha)) + 4*alpha*ti*np.cos(np.pi*ti*(1+alpha))
            den = np.pi*ti*(1 - (4*alpha*ti)**2)
            h[i] = num / (den + 1e-10)

    return h / np.sqrt(np.sum(h**2))

# ============== SPACE CHANNEL MODEL ==============

def free_space_path_loss_db(distance, freq=CARRIER_FREQ):
    """Calculate FSPL in dB"""
    fspl = (4 * np.pi * distance * freq / c) ** 2
    return 10 * np.log10(fspl)

def apply_doppler_shift(signal_in, doppler_hz, sample_rate):
    """Apply Doppler frequency shift"""
    t = np.arange(len(signal_in)) / sample_rate
    shift = np.exp(2j * np.pi * doppler_hz * t)
    return signal_in * shift

def phase_noise(signal_len, phase_noise_std_rad=0.01):
    """Generate phase noise (random walk model)"""
    phase_increments = phase_noise_std_rad * np.random.randn(signal_len)
    phase = np.cumsum(phase_increments)
    return np.exp(1j * phase)

def iq_imbalance(signal_in, gain_imbalance_db=0.1, phase_imbalance_deg=1.0):
    """Apply I/Q imbalance (relevant for QPSK)"""
    g = 10 ** (gain_imbalance_db / 20)
    phi = np.deg2rad(phase_imbalance_deg)

    I = np.real(signal_in)
    Q = np.imag(signal_in)

    I_out = I
    Q_out = g * (np.sin(phi) * I + np.cos(phi) * Q)

    return I_out + 1j * Q_out

def apply_doppler_shift_rf(rf_signal, doppler_hz, sample_rate):
    """Apply Doppler frequency shift to RF signal using Hilbert transform"""
    from scipy.signal import hilbert
    t = np.arange(len(rf_signal)) / sample_rate
    analytic_signal = hilbert(rf_signal)
    phase_shift = np.exp(1j * 2 * np.pi * doppler_hz * t)
    return np.real(analytic_signal * phase_shift)

def phase_noise_rf(signal_len, sample_rate, phase_noise_dbc_hz=-85):
    """Generate phase noise for RF oscillator"""
    effective_noise_bw = SYMBOL_RATE
    phase_var = min(10 ** (phase_noise_dbc_hz / 10) * effective_noise_bw, 0.01)
    phase_std = np.sqrt(phase_var / signal_len)
    phase = np.cumsum(phase_std * np.random.randn(signal_len))
    return np.exp(1j * phase)

def meo_space_channel_rf(rf_signal, distance_km, doppler_hz, snr_db, sample_rate=RF_SAMPLE_RATE):
    """Complete MEO space channel model for RF signal at 2.2 GHz carrier"""
    # Apply Doppler shift to RF signal
    rf_doppler = apply_doppler_shift_rf(rf_signal, doppler_hz, sample_rate)
    # Apply phase noise (oscillator instability)
    from scipy.signal import hilbert
    analytic = hilbert(rf_doppler)
    pn = phase_noise_rf(len(analytic), sample_rate, phase_noise_dbc_hz=-85)
    rf_with_pn = np.real(analytic * pn)
    # Add AWGN (real-valued for RF)
    signal_power = np.mean(rf_with_pn ** 2)
    snr_linear = 10 ** (snr_db / 10)
    noise_power = signal_power / snr_linear
    noise = np.sqrt(noise_power) * np.random.randn(len(rf_with_pn))
    return rf_with_pn + noise

# ============== CARRIER RECOVERY (QPSK) ==============

def costas_loop_qpsk(signal_in, loop_bw=0.01):
    """
    Costas loop for QPSK carrier recovery
    Uses decision-directed phase detector
    """
    phase = 0
    freq = 0
    output = np.zeros_like(signal_in, dtype=complex)

    # Loop filter coefficients
    damping = 0.707
    K1 = 4 * damping * loop_bw
    K2 = 4 * loop_bw ** 2

    for i in range(len(signal_in)):
        # Mix with NCO
        output[i] = signal_in[i] * np.exp(-1j * phase)

        # QPSK phase detector: sign(I)*Q - sign(Q)*I
        I = np.real(output[i])
        Q = np.imag(output[i])
        error = np.sign(I) * Q - np.sign(Q) * I

        # Loop filter
        freq += K2 * error
        phase += K1 * error + freq

    return output

def fourth_power_freq_est(signal_in, sample_rate):
    """
    Fourth power frequency estimation for QPSK
    Removes modulation to estimate carrier offset
    """
    # Raise to 4th power to remove QPSK modulation
    signal_4th = signal_in ** 4

    # FFT to find frequency offset
    N = len(signal_4th)
    spectrum = np.fft.fft(signal_4th)
    freqs = np.fft.fftfreq(N, 1/sample_rate)

    # Find peak (frequency offset is 1/4 of detected)
    peak_idx = np.argmax(np.abs(spectrum))
    freq_offset = freqs[peak_idx] / 4

    return freq_offset

# ============== TIMING RECOVERY ==============

def gardner_ted(signal_in, sps):
    """
    Gardner Timing Error Detector for QPSK
    Works on both I and Q channels
    """
    # Simplified implementation - returns timing error
    errors = []
    for i in range(sps, len(signal_in) - sps, sps):
        # Sample at symbol, mid-point, previous symbol
        curr = signal_in[i]
        mid = signal_in[i - sps//2]
        prev = signal_in[i - sps]

        # Gardner error: real(mid * conj(curr - prev))
        error = np.real(mid * np.conj(curr - prev))
        errors.append(error)

    return np.array(errors)

# ============== RF UPCONVERSION (TX) ==============

def upconvert_to_rf(baseband_signal, carrier_freq=CARRIER_FREQ):
    """Upconvert complex baseband signal to RF at 2.2 GHz carrier"""
    # Upsample baseband signal to RF sample rate
    baseband_upsampled = np.zeros(len(baseband_signal) * RF_OVERSAMPLE_FACTOR, dtype=complex)
    baseband_upsampled[::RF_OVERSAMPLE_FACTOR] = baseband_signal

    # Interpolation filter (anti-imaging)
    interp_filter = np.sinc(np.arange(-32, 32) / RF_OVERSAMPLE_FACTOR)
    interp_filter *= np.hamming(len(interp_filter))
    baseband_interp = np.convolve(baseband_upsampled, interp_filter, mode='same')

    # Quadrature upconversion to simulation IF
    t = np.arange(len(baseband_interp)) / RF_SAMPLE_RATE
    I, Q = np.real(baseband_interp), np.imag(baseband_interp)
    rf_signal = I * np.cos(2*np.pi*SIMULATION_IF*t) - Q * np.sin(2*np.pi*SIMULATION_IF*t)
    return rf_signal, RF_SAMPLE_RATE

# ============== TRANSMITTER ==============

def meo_qpsk_transmitter(data_bits, use_rf_upconversion=True):
    """Complete MEO QPSK transmitter with 2.2 GHz upconversion"""

    # Convolutional encoding (rate 1/2)
    encoded = conv_encode(data_bits)
    # Interleaving
    interleaved = block_interleave(encoded)
    # Scrambling
    scrambled = scramble(interleaved)
    # QPSK modulation (includes S/P conversion)
    symbols = qpsk_modulate_fast(scrambled)
    # Differential encoding for phase ambiguity resolution
    diff_encoded = differential_encode_qpsk(symbols)
    # Upsampling
    upsampled = np.zeros(len(diff_encoded) * SAMPLES_PER_SYMBOL, dtype=complex)
    upsampled[::SAMPLES_PER_SYMBOL] = diff_encoded
    # Pulse shaping (RRC filter)
    num_taps = FILTER_SPAN * SAMPLES_PER_SYMBOL + 1
    rrc = rrc_filter(num_taps, ROLLOFF, SAMPLES_PER_SYMBOL)
    baseband_signal = np.convolve(upsampled, rrc, mode='same')

    if use_rf_upconversion:
        # Upconvert to 2.2 GHz carrier (represented by simulation IF)
        rf_signal, rf_sample_rate = upconvert_to_rf(baseband_signal, CARRIER_FREQ)
        return rf_signal, baseband_signal, diff_encoded, rrc, len(diff_encoded), rf_sample_rate
    else:
        return baseband_signal, baseband_signal, diff_encoded, rrc, len(diff_encoded), BASEBAND_SAMPLE_RATE

# ============== RF DOWNCONVERSION (RX) ==============

def downconvert_from_rf(rf_signal, carrier_freq=CARRIER_FREQ):
    """Downconvert RF signal from 2.2 GHz carrier to complex baseband"""
    t = np.arange(len(rf_signal)) / RF_SAMPLE_RATE
    # Quadrature downconversion from simulation IF
    I_mixed = rf_signal * np.cos(2*np.pi*SIMULATION_IF*t) * 2
    Q_mixed = -rf_signal * np.sin(2*np.pi*SIMULATION_IF*t) * 2
    # Low-pass filter to remove double-frequency components
    nyq = RF_SAMPLE_RATE / 2
    cutoff = BASEBAND_SAMPLE_RATE / 2 * 0.8
    b, a = butter(6, cutoff / nyq, btype='low')
    I_filtered = filtfilt(b, a, I_mixed)
    Q_filtered = filtfilt(b, a, Q_mixed)
    # Decimate back to baseband sample rate
    baseband_signal = (I_filtered + 1j*Q_filtered)[::RF_OVERSAMPLE_FACTOR]
    return baseband_signal

# ============== RECEIVER ==============

def meo_qpsk_receiver(rf_signal, rrc, num_symbols, num_info_bits, doppler_hz=0,
                       use_rf_downconversion=True):
    """Complete MEO QPSK receiver with 2.2 GHz downconversion"""

    if use_rf_downconversion:
        # Downconvert from 2.2 GHz carrier to complex baseband
        baseband = downconvert_from_rf(rf_signal, CARRIER_FREQ)
        # Additional low-pass filtering at baseband
        nyq = BASEBAND_SAMPLE_RATE / 2
        b, a = butter(5, SYMBOL_RATE / nyq, btype='low')
        baseband = filtfilt(b, a, np.real(baseband)) + 1j*filtfilt(b, a, np.imag(baseband))
    else:
        baseband = rf_signal  # Already complex baseband

    # Doppler compensation
    if doppler_hz != 0:
        t = np.arange(len(baseband)) / BASEBAND_SAMPLE_RATE
        baseband = baseband * np.exp(-2j*np.pi*doppler_hz*t)

    # Matched filter FIRST (before carrier recovery)
    filtered = np.convolve(baseband, rrc, mode='same')
    # Downsample to symbol rate
    rx_symbols_raw = filtered[::SAMPLES_PER_SYMBOL][:num_symbols]
    # Carrier recovery (QPSK Costas loop)
    rx_symbols = costas_loop_qpsk(rx_symbols_raw, loop_bw=0.02)
    # AGC normalization
    rx_symbols = rx_symbols / np.sqrt(np.mean(np.abs(rx_symbols)**2)) * np.sqrt(0.5)
    # Differential decoding
    diff_decoded = differential_decode_qpsk(rx_symbols)
    # QPSK demodulation
    rx_bits = qpsk_demodulate(diff_decoded)
    # Descramble
    descrambled = descramble(rx_bits)
    # De-interleave
    deinterleaved = block_deinterleave(descrambled)
    # Viterbi decoding
    decoded = viterbi_decode(deinterleaved, num_info_bits)

    return decoded, rx_symbols

# ============== LINK BUDGET ==============

def calculate_link_budget(distance_km):
    """Calculate link budget for QPSK MEO ISL"""
    distance = distance_km * 1000

    tx_power_dbw = 10 * np.log10(TX_POWER_W)
    eirp = tx_power_dbw + TX_ANTENNA_GAIN_DBI - LINE_LOSS_DB
    fspl = free_space_path_loss_db(distance)
    rx_gain = RX_ANTENNA_GAIN_DBI - LINE_LOSS_DB
    g_over_t = rx_gain - 10 * np.log10(SYSTEM_NOISE_TEMP_K)
    c_n0 = eirp - fspl - 0.5 + g_over_t + 228.6
    # QPSK: 2 bits/symbol, so Eb/N0 = C/N0 - 10*log10(bit_rate)
    eb_n0 = c_n0 - 10 * np.log10(BIT_RATE)

    return {
        'EIRP_dBW': eirp,
        'FSPL_dB': fspl,
        'C_N0_dBHz': c_n0,
        'Eb_N0_dB': eb_n0,
        'Bit_Rate_Mbps': BIT_RATE / 1e6
    }

# ============== BER CALCULATION ==============

def theoretical_qpsk_ber(eb_n0_db):
    """
    Theoretical QPSK BER in AWGN
    Same as BPSK when measured per bit (Eb/N0)
    """
    from scipy.special import erfc
    eb_n0 = 10 ** (eb_n0_db / 10)
    return 0.5 * erfc(np.sqrt(eb_n0))

def theoretical_qpsk_ser(es_n0_db):
    """
    Theoretical QPSK Symbol Error Rate
    Es/N0 = Eb/N0 + 3dB for QPSK (2 bits/symbol)
    """
    from scipy.special import erfc
    es_n0 = 10 ** (es_n0_db / 10)
    return erfc(np.sqrt(es_n0/2)) - 0.25 * erfc(np.sqrt(es_n0/2))**2

def meo_qpsk_isl_simulation(num_bits=10000, distance_km=15000, doppler_hz=5000,
                             use_rf_processing=True):
    """Complete MEO QPSK ISL system simulation with 2.2 GHz RF processing"""

    print(f"MEO QPSK ISL Simulation with 2.2 GHz RF Processing")
    print(f"Carrier Frequency: {CARRIER_FREQ/1e9:.1f} GHz")
    print(f"Distance: {distance_km} km | Doppler: {doppler_hz/1000:.1f} kHz")
    print(f"RF Sample Rate: {RF_SAMPLE_RATE/1e6:.1f} MHz | Simulation IF: {SIMULATION_IF/1e6:.1f} MHz")
    print()

    # Link budget
    budget = calculate_link_budget(distance_km)
    print(f"Link Budget: EIRP={budget['EIRP_dBW']:.1f}dBW, FSPL={budget['FSPL_dB']:.1f}dB, Eb/N0={budget['Eb_N0_dB']:.1f}dB")

    # Generate random data
    tx_bits = np.random.randint(0, 2, num_bits)

    # Transmitter with RF upconversion to 2.2 GHz
    tx_rf, tx_baseband, tx_symbols, rrc, num_tx_symbols, rf_sample_rate = \
        meo_qpsk_transmitter(tx_bits, use_rf_upconversion=use_rf_processing)

    if use_rf_processing:
        # RF Channel at 2.2 GHz carrier
        rx_rf = meo_space_channel_rf(tx_rf, distance_km, doppler_hz,
                                      budget['Eb_N0_dB'], rf_sample_rate)
        # Receiver with RF downconversion from 2.2 GHz
        rx_bits, rx_symbols = meo_qpsk_receiver(
            rx_rf, rrc, num_tx_symbols, num_bits, doppler_hz,
            use_rf_downconversion=True)
    else:
        rx_bits, rx_symbols = meo_qpsk_receiver(
            tx_baseband, rrc, num_tx_symbols, num_bits, doppler_hz,
            use_rf_downconversion=False)

    # Calculate BER
    min_len = min(len(tx_bits), len(rx_bits))
    bit_errors = np.sum(tx_bits[:min_len] != rx_bits[:min_len])
    ber = bit_errors / min_len

    print(f"Results: {bit_errors} errors in {min_len} bits, BER={ber:.2e}")

    return {'tx_bits': tx_bits, 'rx_bits': rx_bits, 'ber': ber,
            'tx_symbols': tx_symbols, 'rx_symbols': rx_symbols,
            'carrier_freq_ghz': CARRIER_FREQ/1e9, 'rf_sample_rate': rf_sample_rate}

QPSK Space Environment Advantages
  • Double Spectral Efficiency: 2 bits/symbol vs BPSK's 1 bit/symbol
  • Same Eb/N0: Identical BER to BPSK when measured per bit
  • Constant Envelope: No amplitude variations - ideal for nonlinear amplifiers
  • Industry Standard: DVB-S2, GPS L2C, CCSDS recommendations
  • Bandwidth Efficient: Same bandwidth as BPSK, twice the throughput
  • I/Q Processing: Compatible with modern SDR architectures
Key Design Choices
  • RF Processing: Full upconversion/downconversion at 2.2 GHz (160 MHz sim IF)
  • Gray Coding: Adjacent symbols differ by 1 bit, minimizes BER
  • Conv. Code R=1/2, K=7: ~5 dB coding gain, NASA heritage
  • Differential Encoding: Resolves 90° phase ambiguity in QPSK
  • Hilbert Transform: Accurate RF Doppler shift via analytic signal
  • Costas Loop: QPSK carrier recovery with sign(I)Q-sign(Q)I detector
