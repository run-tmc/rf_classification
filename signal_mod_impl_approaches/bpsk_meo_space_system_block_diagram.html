# ============== BPSK MODULATION ==============

def bpsk_modulate(bits):
    """
    BPSK modulation: map bits to symbols
    bit 0 -> -1 (phase 180°)
    bit 1 -> +1 (phase 0°)
    """
    return 2 * bits.astype(float) - 1  # Maps 0->-1, 1->+1

def bpsk_demodulate(symbols):
    """
    BPSK demodulation: hard decision
    negative -> bit 0
    positive -> bit 1
    """
    return (np.real(symbols) > 0).astype(int)

# ============== DIFFERENTIAL ENCODING ==============

def differential_encode(bits):
    """Differential encoding to resolve phase ambiguity"""
    encoded = np.zeros(len(bits) + 1, dtype=int)
    encoded[0] = 1  # Reference bit
    for i in range(len(bits)):
        encoded[i+1] = encoded[i] ^ bits[i]
    return encoded[1:]

def differential_decode(bits):
    """
    Differential decoding

    Reverses the differential encoding by prepending the reference bit (1)
    and computing XOR between adjacent bits to recover the original data.
    """
    # Prepend the reference bit used in encoding
    bits_with_ref = np.concatenate([[1], bits])

    # Decode: original_bit[i] = encoded[i] XOR encoded[i+1]
    decoded = np.zeros(len(bits), dtype=int)
    for i in range(len(bits)):
        decoded[i] = bits_with_ref[i] ^ bits_with_ref[i+1]
    return decoded

# ============== CONVOLUTIONAL ENCODER (Rate 1/2, K=7) ==============

def conv_encode(bits, g1=0o171, g2=0o133):
    """
    Convolutional encoder - NASA standard rate 1/2, K=7
    g1 = 171 octal (1111001)
    g2 = 133 octal (1011011)
    """
    K = 7  # Constraint length
    state = 0
    encoded = []

    for bit in bits:
        state = ((state << 1) | bit) & (2**K - 1)

        # Output 1: XOR of bits selected by g1
        out1 = bin(state & g1).count('1') % 2
        # Output 2: XOR of bits selected by g2
        out2 = bin(state & g2).count('1') % 2

        encoded.extend([out1, out2])

    # Flush encoder with zeros
    for _ in range(K - 1):
        state = (state << 1) & (2**K - 1)
        out1 = bin(state & g1).count('1') % 2
        out2 = bin(state & g2).count('1') % 2
        encoded.extend([out1, out2])

    return np.array(encoded)

def viterbi_decode(received, num_info_bits, g1=0o171, g2=0o133):
    """
    Viterbi decoder for rate 1/2, K=7 convolutional code
    Hard decision decoding
    """
    K = 7
    num_states = 2 ** (K - 1)

    # Initialize path metrics
    path_metric = np.full(num_states, np.inf)
    path_metric[0] = 0

    # Survivor paths
    survivors = np.zeros((len(received)//2, num_states), dtype=int)

    # Process received bits in pairs
    for i in range(0, len(received), 2):
        rx_bits = received[i:i+2]
        new_path_metric = np.full(num_states, np.inf)

        for state in range(num_states):
            for input_bit in [0, 1]:
                # Calculate next state
                next_state = ((state << 1) | input_bit) & (num_states - 1)

                # Calculate expected output
                full_state = (state << 1) | input_bit
                exp_out1 = bin(full_state & g1).count('1') % 2
                exp_out2 = bin(full_state & g2).count('1') % 2

                # Hamming distance
                branch_metric = (rx_bits[0] != exp_out1) + (rx_bits[1] != exp_out2)

                # Update path metric
                candidate_metric = path_metric[state] + branch_metric
                if candidate_metric < new_path_metric[next_state]:
                    new_path_metric[next_state] = candidate_metric
                    survivors[i//2, next_state] = state

        path_metric = new_path_metric

    # Traceback
    state = 0  # Assume terminated to zero state
    decoded = []
    for i in range(len(received)//2 - 1, -1, -1):
        prev_state = survivors[i, state]
        decoded_bit = (state >> (K - 2)) & 1
        decoded.append(decoded_bit)
        state = prev_state

    decoded = np.array(decoded[::-1])
    return decoded[:num_info_bits]

# ============== INTERLEAVER ==============

def block_interleave(bits, rows=32):
    """Block interleaver for burst error protection"""
    cols = int(np.ceil(len(bits) / rows))
    padded = np.zeros(rows * cols, dtype=int)
    padded[:len(bits)] = bits
    matrix = padded.reshape(rows, cols)
    return matrix.T.flatten()

def block_deinterleave(bits, rows=32):
    """Block de-interleaver"""
    cols = int(np.ceil(len(bits) / rows))
    matrix = bits[:rows*cols].reshape(cols, rows)
    return matrix.T.flatten()

# ============== SCRAMBLER ==============

def lfsr_sequence(length, seed=0x1ACFFC1D):
    """Generate LFSR pseudo-random sequence"""
    state = seed
    seq = np.zeros(length, dtype=int)
    for i in range(length):
        seq[i] = state & 1
        newbit = ((state >> 0) ^ (state >> 2) ^ (state >> 3) ^ (state >> 5)) & 1
        state = (state >> 1) | (newbit << 31)
    return seq

def scramble(bits):
    """Scramble bits with LFSR sequence"""
    pn = lfsr_sequence(len(bits))
    return (bits ^ pn).astype(int)

def descramble(bits):
    """Descramble bits"""
    return scramble(bits)

# ============== PULSE SHAPING ==============

def rrc_filter(num_taps, alpha, sps):
    """Root Raised Cosine filter design"""
    t = np.arange(-num_taps//2, num_taps//2 + 1) / sps
    h = np.zeros_like(t, dtype=float)

    for i, ti in enumerate(t):
        if ti == 0:
            h[i] = 1 + alpha * (4/np.pi - 1)
        elif abs(abs(ti) - 1/(4*alpha)) < 1e-10:
            h[i] = alpha/np.sqrt(2) * ((1+2/np.pi)*np.sin(np.pi/(4*alpha)) +
                                         (1-2/np.pi)*np.cos(np.pi/(4*alpha)))
        else:
            num = np.sin(np.pi*ti*(1-alpha)) + 4*alpha*ti*np.cos(np.pi*ti*(1+alpha))
            den = np.pi*ti*(1 - (4*alpha*ti)**2)
            h[i] = num / (den + 1e-10)

    return h / np.sqrt(np.sum(h**2))

# ============== SPACE CHANNEL MODEL ==============

def free_space_path_loss_db(distance, freq=CARRIER_FREQ):
    """Calculate Free Space Path Loss in dB at the RF carrier frequency"""
    fspl = (4 * np.pi * distance * freq / c) ** 2
    return 10 * np.log10(fspl)

def calculate_doppler_shift(relative_velocity_mps, carrier_freq=CARRIER_FREQ):
    """Calculate Doppler shift at the 2.2 GHz carrier frequency"""
    return carrier_freq * (relative_velocity_mps / c)

def apply_doppler_shift(signal_in, doppler_hz, sample_rate):
    """Apply Doppler frequency shift to the signal"""
    t = np.arange(len(signal_in)) / sample_rate
    shift = np.exp(2j * np.pi * doppler_hz * t)
    return signal_in * shift

def phase_noise(signal_len, phase_noise_std_rad=0.01):
    """Generate phase noise (random walk model for LO instability)"""
    phase_increments = phase_noise_std_rad * np.random.randn(signal_len)
    phase = np.cumsum(phase_increments)
    return np.exp(1j * phase)

def atmospheric_attenuation_db(distance_km, freq=CARRIER_FREQ):
    """Calculate atmospheric attenuation for space-to-space links"""
    scintillation_loss = 0.1  # dB, typical for S-band
    return scintillation_loss

def apply_rf_channel_effects(rf_signal, distance_km, doppler_hz, carrier_freq=CARRIER_FREQ):
    """Apply RF propagation effects at the 2.2 GHz carrier frequency"""
    distance = distance_km * 1000
    prop_delay_sec = distance / c
    prop_delay_samples = int(prop_delay_sec * SAMPLE_RATE)

    # Apply Doppler shift at the carrier frequency
    sig = apply_doppler_shift(rf_signal, doppler_hz, SAMPLE_RATE)

    # Apply phase noise (models LO instability)
    sig = sig * phase_noise(len(sig))

    # Apply propagation delay
    if prop_delay_samples > 0 and prop_delay_samples < len(sig):
        sig = np.roll(sig, prop_delay_samples)
        sig[:prop_delay_samples] = 0

    return sig

def meo_space_channel(signal_in, distance_km, doppler_hz, snr_db):
    """Complete MEO space channel model for 2.2 GHz RF carrier"""
    # Apply RF channel effects at 2.2 GHz carrier
    sig = apply_rf_channel_effects(signal_in, distance_km, doppler_hz, CARRIER_FREQ)

    # Add AWGN based on link budget SNR
    signal_power = np.mean(np.abs(sig) ** 2)
    snr_linear = 10 ** (snr_db / 10)
    noise_power = signal_power / snr_linear
    noise = np.sqrt(noise_power/2) * (np.random.randn(len(sig)) + 1j*np.random.randn(len(sig)))

    return sig + noise

# ============== CARRIER RECOVERY ==============

def costas_loop_bpsk(signal_in, loop_bw=0.01):
    """Costas loop for BPSK carrier recovery"""
    phase = 0
    freq = 0
    output = np.zeros_like(signal_in, dtype=complex)

    # Loop filter coefficients
    damping = 0.707
    K1 = 4 * damping * loop_bw
    K2 = 4 * loop_bw ** 2

    for i in range(len(signal_in)):
        # Mix with NCO
        output[i] = signal_in[i] * np.exp(-1j * phase)

        # BPSK phase detector: sign(I) * Q
        error = np.sign(np.real(output[i])) * np.imag(output[i])

        # Loop filter
        freq += K2 * error
        phase += K1 * error + freq

    return output

# ============== TRANSMITTER ==============

def baseband_processing_tx(data_bits):
    """Transmitter baseband signal processing chain"""
    # Convolutional encoding (rate 1/2)
    encoded = conv_encode(data_bits)
    # Interleaving
    interleaved = block_interleave(encoded)
    # Scrambling
    scrambled = scramble(interleaved)
    # Differential encoding
    diff_encoded = differential_encode(scrambled)
    # BPSK modulation
    symbols = bpsk_modulate(diff_encoded)
    # Upsampling
    upsampled = np.zeros(len(symbols) * SAMPLES_PER_SYMBOL, dtype=complex)
    upsampled[::SAMPLES_PER_SYMBOL] = symbols
    # Pulse shaping with Root Raised Cosine filter
    num_taps = FILTER_SPAN * SAMPLES_PER_SYMBOL + 1
    rrc = rrc_filter(num_taps, ROLLOFF, SAMPLES_PER_SYMBOL)
    baseband_signal = np.convolve(upsampled, rrc, mode='same')
    return baseband_signal, symbols, rrc, len(diff_encoded)

def upconvert_to_if(baseband_signal, if_freq=IF_FREQ, sample_rate=SAMPLE_RATE):
    """Upconvert baseband I/Q signal to intermediate frequency (IF)"""
    t = np.arange(len(baseband_signal)) / sample_rate
    if_signal = baseband_signal * np.exp(1j * 2 * np.pi * if_freq * t)
    return if_signal

def upconvert_if_to_rf(if_signal, carrier_freq=CARRIER_FREQ):
    """Upconvert IF signal to 2.2 GHz RF carrier frequency"""
    # Conceptual upconversion - channel effects applied at carrier freq
    rf_signal = if_signal.copy()
    return rf_signal

def meo_bpsk_transmitter(data_bits):
    """Complete MEO BPSK satellite transmitter chain with RF upconversion"""
    # Step 1: Baseband digital signal processing
    baseband_signal, symbols, rrc, num_symbols = baseband_processing_tx(data_bits)
    # Step 2: Upconvert to IF (20 MHz simulation, 70 MHz hardware)
    if_signal = upconvert_to_if(baseband_signal, IF_FREQ, SAMPLE_RATE)
    # Step 3: Upconvert to RF carrier (2.2 GHz)
    rf_signal = upconvert_if_to_rf(if_signal, CARRIER_FREQ)
    return rf_signal, symbols, rrc, num_symbols

# ============== RECEIVER ==============

def downconvert_from_rf_carrier(rf_signal, carrier_freq=CARRIER_FREQ):
    """Downconvert from 2.2 GHz RF carrier to IF"""
    if_signal = rf_signal.copy()
    return if_signal

def downconvert_to_baseband(if_signal, if_freq=IF_FREQ, sample_rate=SAMPLE_RATE):
    """Downconvert IF signal to complex baseband (I/Q demodulation)"""
    t = np.arange(len(if_signal)) / sample_rate
    baseband_signal = if_signal * np.exp(-1j * 2 * np.pi * if_freq * t)
    return baseband_signal

def doppler_compensation(signal_in, doppler_hz, sample_rate=SAMPLE_RATE):
    """Compensate for Doppler shift at the carrier frequency"""
    return apply_doppler_shift(signal_in, -doppler_hz, sample_rate)

def baseband_processing_rx(baseband_signal, rrc, num_symbols, num_info_bits):
    """Receiver baseband signal processing chain"""
    # Carrier recovery (Costas loop)
    recovered = costas_loop_bpsk(baseband_signal)
    # Matched filter
    filtered = np.convolve(recovered, rrc, mode='same')
    # Symbol timing recovery and downsample
    offset = SAMPLES_PER_SYMBOL // 2 - 1
    rx_symbols = filtered[offset::SAMPLES_PER_SYMBOL][:num_symbols]
    # AGC normalization
    rx_symbols = rx_symbols / np.std(np.real(rx_symbols))
    # BPSK demodulation (hard decision)
    rx_bits = bpsk_demodulate(rx_symbols)
    # Differential decoding
    diff_decoded = differential_decode(rx_bits)
    # Calculate expected size for de-interleaver
    rows = 32
    expected_encoded_len = 2 * (num_info_bits + 6)
    cols = int(np.ceil(expected_encoded_len / rows))
    expected_interleaved_len = rows * cols
    # Pad descrambled bits
    descrambled = descramble(diff_decoded)
    if len(descrambled) < expected_interleaved_len:
        descrambled = np.pad(descrambled, (0, expected_interleaved_len - len(descrambled)), 'constant')
    else:
        descrambled = descrambled[:expected_interleaved_len]
    # De-interleave
    deinterleaved = block_deinterleave(descrambled, rows)
    # Viterbi decoding
    decoded_bits = viterbi_decode(deinterleaved, num_info_bits)
    return decoded_bits, rx_symbols

def meo_bpsk_receiver(rf_signal, rrc, num_symbols, num_info_bits, doppler_hz=0):
    """Complete MEO BPSK satellite receiver chain with RF downconversion"""
    # Step 1: RF Frontend - Downconvert from 2.2 GHz to IF
    if_signal = downconvert_from_rf_carrier(rf_signal, CARRIER_FREQ)
    # Step 2: IF to Baseband downconversion (I/Q demodulation)
    baseband_signal = downconvert_to_baseband(if_signal, IF_FREQ, SAMPLE_RATE)
    # Step 3: Doppler compensation
    if doppler_hz != 0:
        baseband_signal = doppler_compensation(baseband_signal, doppler_hz, SAMPLE_RATE)
    # Step 4: Baseband signal processing
    decoded_bits, rx_symbols = baseband_processing_rx(baseband_signal, rrc, num_symbols, num_info_bits)
    return decoded_bits, rx_symbols

# ============== LINK BUDGET ==============

def calculate_link_budget(distance_km):
    """Calculate link budget for BPSK MEO ISL"""
    distance = distance_km * 1000

    tx_power_dbw = 10 * np.log10(TX_POWER_W)
    eirp = tx_power_dbw + TX_ANTENNA_GAIN_DBI - LINE_LOSS_DB
    fspl = free_space_path_loss_db(distance)
    rx_gain = RX_ANTENNA_GAIN_DBI - LINE_LOSS_DB
    g_over_t = rx_gain - 10 * np.log10(SYSTEM_NOISE_TEMP_K)
    c_n0 = eirp - fspl - 0.5 + g_over_t + 228.6
    eb_n0 = c_n0 - 10 * np.log10(SYMBOL_RATE)  # BPSK: 1 bit/symbol

    return {
        'EIRP_dBW': eirp,
        'FSPL_dB': fspl,
        'C_N0_dBHz': c_n0,
        'Eb_N0_dB': eb_n0
    }

# ============== BER CALCULATION ==============

def theoretical_bpsk_ber(eb_n0_db):
    """Theoretical BPSK BER in AWGN"""
    from scipy.special import erfc
    eb_n0 = 10 ** (eb_n0_db / 10)
    return 0.5 * erfc(np.sqrt(eb_n0))

BPSK Space Environment Advantages
  • Maximum Noise Immunity: Best BER vs Eb/N0 of all PSK schemes
  • Constant Envelope: No amplitude variations - ideal for nonlinear amplifiers
  • Simple Implementation: Sign detection receiver, minimal complexity
  • Robust Synchronization: Costas loop works well at low SNR
  • Heritage: Used in GPS L1 C/A, deep space missions, military satcom
  • Lower Doppler Sensitivity: Wider tracking range at S-band
Key Design Choices
  • Conv. Code R=1/2, K=7: ~5 dB coding gain, NASA heritage
  • Viterbi Decoder: Hard-decision decoding with traceback
  • Differential Encoding: Resolves 180° phase ambiguity
  • S-Band (2.2 GHz): Lower FSPL, less Doppler than Ka-band
  • IF Architecture: 70 MHz IF (hw), 20 MHz simulation IF
  • RRC α=0.35: Span 10 symbols, ISI minimization
  • Costas Loop: BPSK carrier recovery, ζ=0.707
