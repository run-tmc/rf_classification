# 64-QAM constellation levels
QAM64_LEVELS = np.array([-7, -5, -3, -1, 1, 3, 5, 7])
QAM64_NORM = np.sqrt(42)  # Normalization factor for unit average power

# Gray-coded 64-QAM mapping (3 bits -> level)
GRAY_MAP = {0: -7, 1: -5, 3: -3, 2: -1, 6: 1, 7: 3, 5: 5, 4: 7GRAY_DEMAP = {-7: 0, -5: 1, -3: 3, -1: 2, 1: 6, 3: 7, 5: 5, 7: 4# ============== 64-QAM MODULATION ==============

def qam64_modulate(bits):
    """
    64-QAM modulation: map 6-bit groups to complex symbols
    Gray coding: adjacent symbols differ by 1 bit
    """
    # Pad to multiple of 6 bits
    pad_len = (6 - len(bits) % 6) % 6
    if pad_len > 0:
        bits = np.append(bits, np.zeros(pad_len, dtype=int))

    # Reshape into 6-bit groups
    bit_groups = bits.reshape(-1, 6)

    symbols = []
    for group in bit_groups:
        # Extract I bits (b5, b4, b3) and Q bits (b2, b1, b0)
        i_bits = group[0] * 4 + group[1] * 2 + group[2]  # b5b4b3
        q_bits = group[3] * 4 + group[4] * 2 + group[5]  # b2b1b0

        # Gray-coded mapping
        i_level = GRAY_MAP[i_bits]
        q_level = GRAY_MAP[q_bits]

        # Create normalized complex symbol
        symbol = (i_level + 1j * q_level) / QAM64_NORM
        symbols.append(symbol)

    return np.array(symbols)

def qam64_demodulate(symbols):
    """
    64-QAM demodulation: hard decision on I/Q levels
    Returns 6-bit groups based on nearest constellation point
    """
    bits = []
    thresholds = [-6, -4, -2, 0, 2, 4, 6]

    for s in symbols:
        # Denormalize
        s_denorm = s * QAM64_NORM

        # Slice I and Q components
        i_idx = np.digitize(np.real(s_denorm), thresholds)
        q_idx = np.digitize(np.imag(s_denorm), thresholds)

        i_level = QAM64_LEVELS[i_idx]
        q_level = QAM64_LEVELS[q_idx]

        # Gray demap to bits
        i_bits = GRAY_DEMAP[i_level]
        q_bits = GRAY_DEMAP[q_level]

        # Convert to individual bits
        bits.extend([(i_bits >> 2) & 1, (i_bits >> 1) & 1, i_bits & 1,
                     (q_bits >> 2) & 1, (q_bits >> 1) & 1, q_bits & 1])

    return np.array(bits)

def qam64_soft_demodulate(symbols, noise_var):
    """
    Soft 64-QAM demodulation: compute LLRs for LDPC decoder
    """
    llr = np.zeros(len(symbols) * 6)

    # Generate all 64 constellation points
    constellation = []
    bit_patterns = []
    for i in range(64):
        i_bits = (i >> 3) & 7
        q_bits = i & 7
        i_level = GRAY_MAP[i_bits]
        q_level = GRAY_MAP[q_bits]
        constellation.append((i_level + 1j * q_level) / QAM64_NORM)
        bit_patterns.append([(i >> 5) & 1, (i >> 4) & 1, (i >> 3) & 1,
                             (i >> 2) & 1, (i >> 1) & 1, i & 1])

    constellation = np.array(constellation)
    bit_patterns = np.array(bit_patterns)

    for i, s in enumerate(symbols):
        distances = np.abs(s - constellation) ** 2
        for b in range(6):
            mask_0 = bit_patterns[:, b] == 0
            mask_1 = bit_patterns[:, b] == 1
            min_d0 = np.min(distances[mask_0])
            min_d1 = np.min(distances[mask_1])
            llr[6*i + b] = (min_d1 - min_d0) / (2 * noise_var)

    return llr

# ============== LDPC ENCODER/DECODER ==============

def ldpc_encode(bits, code_rate=0.75):
    """Simplified LDPC encoder (adds parity bits)"""
    k = len(bits)
    n = int(k / code_rate)
    parity_len = n - k
    parity = np.zeros(parity_len, dtype=int)
    for i in range(parity_len):
        parity[i] = np.sum(bits[i::parity_len]) % 2
    return np.concatenate([bits, parity])

def ldpc_decode(bits, code_rate=0.75):
    """Simplified LDPC decoder (extracts data bits)"""
    k = int(len(bits) * code_rate)
    return bits[:k]

# ============== INTERLEAVER ==============

def block_interleave(bits, rows=64):
    """Block interleaver for burst error protection"""
    cols = int(np.ceil(len(bits) / rows))
    padded = np.zeros(rows * cols, dtype=int)
    padded[:len(bits)] = bits
    matrix = padded.reshape(rows, cols)
    return matrix.T.flatten()

def block_deinterleave(bits, rows=64):
    """Block de-interleaver"""
    cols = int(np.ceil(len(bits) / rows))
    matrix = bits[:rows*cols].reshape(cols, rows)
    return matrix.T.flatten()

# ============== SCRAMBLER ==============

def lfsr_sequence(length, seed=0x1ACFFC1D):
    """Generate LFSR pseudo-random sequence"""
    state = seed
    seq = np.zeros(length, dtype=int)
    for i in range(length):
        seq[i] = state & 1
        newbit = ((state >> 0) ^ (state >> 2) ^ (state >> 3) ^ (state >> 5)) & 1
        state = (state >> 1) | (newbit << 31)
    return seq

def scramble(bits):
    pn = lfsr_sequence(len(bits))
    return (bits ^ pn).astype(int)

def descramble(bits):
    return scramble(bits)

# ============== PULSE SHAPING ==============

def rrc_filter(num_taps, alpha, sps):
    """Root Raised Cosine filter design"""
    t = np.arange(-num_taps//2, num_taps//2 + 1) / sps
    h = np.zeros_like(t, dtype=float)
    for i, ti in enumerate(t):
        if ti == 0:
            h[i] = 1 + alpha * (4/np.pi - 1)
        elif abs(abs(ti) - 1/(4*alpha)) < 1e-10:
            h[i] = alpha/np.sqrt(2) * ((1+2/np.pi)*np.sin(np.pi/(4*alpha)) +
                                         (1-2/np.pi)*np.cos(np.pi/(4*alpha)))
        else:
            num = np.sin(np.pi*ti*(1-alpha)) + 4*alpha*ti*np.cos(np.pi*ti*(1+alpha))
            den = np.pi*ti*(1 - (4*alpha*ti)**2)
            h[i] = num / (den + 1e-10)
    return h / np.sqrt(np.sum(h**2))

# ============== CREST FACTOR REDUCTION ==============

def cfr_clip(signal_in, clip_ratio=0.98):
    """
    Crest Factor Reduction via clipping for 64-QAM PAPR control
    Use higher clip ratio (0.98) to preserve 64-QAM constellation integrity
    """
    peak = np.max(np.abs(signal_in))
    threshold = clip_ratio * peak
    amplitude = np.abs(signal_in)
    phase = np.angle(signal_in)
    clipped_amp = np.minimum(amplitude, threshold)
    return clipped_amp * np.exp(1j * phase)

# ============== DIGITAL PREDISTORTION ==============

def digital_predistort(signal_in, backoff_db=8):
    """Digital predistortion - 8 dB back-off for 64-QAM"""
    backoff_linear = 10 ** (-backoff_db / 20)
    return signal_in * backoff_linear

# ============== RF UP-CONVERSION (TRANSMITTER) ==============

def upconvert_to_rf(baseband_signal, sample_rate_bb=BASEBAND_SAMPLE_RATE):
    """
    Upconvert baseband to RF carrier (27 GHz via 100 MHz IF simulation)
    """
    # Upsample 10x for RF representation
    upsampled = np.zeros(len(baseband_signal) * RF_OVERSAMPLE, dtype=complex)
    upsampled[::RF_OVERSAMPLE] = baseband_signal

    # Anti-imaging lowpass filter
    nyq = RF_SAMPLE_RATE / 2
    b_lp, a_lp = butter(5, sample_rate_bb/2 / nyq, btype='low')
    upsampled_i = filtfilt(b_lp, a_lp, np.real(upsampled)) * RF_OVERSAMPLE
    upsampled_q = filtfilt(b_lp, a_lp, np.imag(upsampled)) * RF_OVERSAMPLE
    upsampled_filtered = upsampled_i + 1j * upsampled_q

    # Upconvert to IF (representing 27 GHz carrier)
    t = np.arange(len(upsampled_filtered)) / RF_SAMPLE_RATE
    carrier_i = np.cos(2 * np.pi * IF_FREQ * t)
    carrier_q = -np.sin(2 * np.pi * IF_FREQ * t)

    # Quadrature upconversion: RF = I*cos(wt) - Q*sin(wt)
    rf_signal = np.real(upsampled_filtered) * carrier_i + np.imag(upsampled_filtered) * carrier_q
    return rf_signal, RF_SAMPLE_RATE

# ============== RF DOWN-CONVERSION (RECEIVER) ==============

def downconvert_to_baseband(rf_signal, rf_sample_rate=RF_SAMPLE_RATE, lo_freq_offset=0.0):
    """
    Downconvert RF signal to baseband using quadrature demodulation
    Returns: baseband_signal, intermediate_signals dict
    """
    t = np.arange(len(rf_signal)) / rf_sample_rate

    # LO with Doppler compensation
    lo_freq = IF_FREQ + lo_freq_offset
    lo_i = 2 * np.cos(2 * np.pi * lo_freq * t)
    lo_q = -2 * np.sin(2 * np.pi * lo_freq * t)

    # Quadrature downconversion
    i_mixed = rf_signal * lo_i
    q_mixed = rf_signal * lo_q

    # Lowpass filter to remove 2*IF component
    nyq = rf_sample_rate / 2
    b_lp, a_lp = butter(5, BASEBAND_SAMPLE_RATE/2 / nyq, btype='low')
    i_baseband = filtfilt(b_lp, a_lp, i_mixed)
    q_baseband = filtfilt(b_lp, a_lp, q_mixed)
    baseband_complex = i_baseband + 1j * q_baseband

    # Store intermediate signals for analysis
    intermediate_signals = {
        'after_downconv': i_mixed + 1j * q_mixed,
        'after_lpf': baseband_complex,
        'time': t
    }

    # Downsample to baseband rate
    baseband_signal = baseband_complex[::RF_OVERSAMPLE]
    return baseband_signal, intermediate_signals

# ============== SPACE CHANNEL MODEL ==============

def free_space_path_loss_db(distance, freq=CARRIER_FREQ):
    """Calculate FSPL in dB"""
    fspl = (4 * np.pi * distance * freq / c) ** 2
    return 10 * np.log10(fspl)

def apply_doppler_shift(signal_in, doppler_hz, sample_rate):
    """Apply Doppler frequency shift"""
    t = np.arange(len(signal_in)) / sample_rate
    shift = np.exp(2j * np.pi * doppler_hz * t)
    return signal_in * shift

def phase_noise(signal_len, phase_noise_std_rad=0.003):
    """
    Generate phase noise - tighter specs for 64-QAM
    """
    phase_increments = phase_noise_std_rad * np.random.randn(signal_len)
    phase = np.cumsum(phase_increments)
    return np.exp(1j * phase)

def iq_imbalance(signal_in, gain_imb_db=0.1, phase_imb_deg=0.3):
    """
    Apply I/Q imbalance - tighter specs for 64-QAM
    """
    g = 10 ** (gain_imb_db / 20)
    phi = np.deg2rad(phase_imb_deg)
    i = np.real(signal_in)
    q = np.imag(signal_in)
    i_out = i
    q_out = g * (np.sin(phi) * i + np.cos(phi) * q)
    return i_out + 1j * q_out

def meo_rf_space_channel(rf_signal, rf_sample_rate, distance_km, doppler_hz, snr_db):
    """
    MEO space channel model for RF carrier signal at 27 GHz
    Includes: FSPL, Doppler shift, phase noise, band-limited AWGN
    """
    from scipy.signal import hilbert
    n_samples = len(rf_signal)
    t = np.arange(n_samples) / rf_sample_rate

    # Scale Doppler to IF representation
    doppler_if = doppler_hz * (IF_FREQ / CARRIER_FREQ)

    # Apply Doppler via analytic signal
    analytic_sig = hilbert(rf_signal)
    freq_shift = np.exp(1j * 2 * np.pi * doppler_if * t)
    sig_shifted = np.real(analytic_sig * freq_shift)

    # Phase noise (space-grade oscillator for 64-QAM)
    rf_phase_noise_std = 0.0002  # rad/sqrt(sample)
    phase_noise_cumulative = np.cumsum(rf_phase_noise_std * np.random.randn(n_samples))
    phase_noise_cumulative = np.clip(phase_noise_cumulative, -0.1, 0.1)
    analytic_with_pn = hilbert(sig_shifted) * np.exp(1j * phase_noise_cumulative)
    sig_with_pn = np.real(analytic_with_pn)

    # Add band-limited thermal noise at RF
    signal_power = np.mean(sig_with_pn ** 2)
    snr_linear = 10 ** (snr_db / 10)
    noise_power = signal_power / snr_linear
    rf_noise = np.sqrt(noise_power) * np.random.randn(n_samples)

    # Bandpass filter noise to signal bandwidth
    bw = SYMBOL_RATE * (1 + ROLLOFF) * 2
    nyq = rf_sample_rate / 2
    low = max(0.01, (IF_FREQ - bw/2) / nyq)
    high = min(0.99, (IF_FREQ + bw/2) / nyq)
    b_bp, a_bp = butter(3, [low, high], btype='band')
    rf_noise_filtered = filtfilt(b_bp, a_bp, rf_noise)
    rf_noise_filtered *= np.sqrt(noise_power / (np.mean(rf_noise_filtered**2) + 1e-20))

    return sig_with_pn + rf_noise_filtered

# ============== CARRIER RECOVERY ==============

def carrier_phase_recovery_symbols(symbols, loop_bw=0.02):
    """
    Decision-directed carrier phase and frequency recovery at symbol rate.
    Second-order PLL to track both phase offset and frequency offset (Doppler).
    """
    phase = 0
    freq = 0
    output = np.zeros_like(symbols, dtype=complex)
    thresholds = np.array([-6, -4, -2, 0, 2, 4, 6]) / QAM64_NORM

    # Second-order loop filter coefficients
    damping = 0.707
    K1 = 2 * damping * loop_bw  # Proportional gain
    K2 = loop_bw ** 2            # Integral gain

    for i in range(len(symbols)):
        output[i] = symbols[i] * np.exp(-1j * phase)

        # Make hard decision
        i_idx = np.clip(np.digitize(np.real(output[i]), thresholds), 0, 7)
        q_idx = np.clip(np.digitize(np.imag(output[i]), thresholds), 0, 7)
        i_dec = QAM64_LEVELS[i_idx] / QAM64_NORM
        q_dec = QAM64_LEVELS[q_idx] / QAM64_NORM
        dec_symbol = i_dec + 1j * q_dec

        # Phase error (ML estimate)
        phase_error = np.angle(output[i] * np.conj(dec_symbol))
        phase_error = np.clip(phase_error, -np.pi/4, np.pi/4)

        # Update frequency (integral) and phase (proportional + integral)
        freq += K2 * phase_error
        freq = np.clip(freq, -0.1, 0.1)
        phase += K1 * phase_error + freq

    return output

# ============== TRANSMITTER ==============

def meo_64qam_transmitter(data_bits, enable_rf_upconversion=True):
    """
    Complete MEO 64-QAM satellite transmitter chain with RF up-conversion
    """
    # Digital baseband processing
    encoded = ldpc_encode(data_bits, code_rate=0.75)
    interleaved = block_interleave(encoded)
    scrambled = scramble(interleaved)
    symbols = qam64_modulate(scrambled)

    # Pulse shaping
    upsampled = np.zeros(len(symbols) * SAMPLES_PER_SYMBOL, dtype=complex)
    upsampled[::SAMPLES_PER_SYMBOL] = symbols
    num_taps = FILTER_SPAN * SAMPLES_PER_SYMBOL + 1
    rrc = rrc_filter(num_taps, ROLLOFF, SAMPLES_PER_SYMBOL)
    shaped = np.convolve(upsampled, rrc, mode='same')

    # CFR and predistortion
    cfr_signal = cfr_clip(shaped)
    predistorted = digital_predistort(cfr_signal, OUTPUT_BACKOFF_DB)
    baseband_signal = predistorted.copy()

    if enable_rf_upconversion:
        # RF up-conversion to 27 GHz carrier (via IF)
        rf_signal, rf_sample_rate = upconvert_to_rf(predistorted)
        return rf_signal, symbols, rrc, rf_sample_rate, baseband_signal
    else:
        return predistorted, symbols, rrc, BASEBAND_SAMPLE_RATE, baseband_signal

# ============== RECEIVER ==============

def meo_64qam_receiver(rx_signal, rrc, num_symbols, num_info_bits, doppler_hz=0,
                       rf_input=True, rf_sample_rate=RF_SAMPLE_RATE):
    """
    Complete MEO 64-QAM satellite receiver chain with RF down-conversion
    """
    if rf_input:
        # RF Front-End: Downconvert with Doppler compensation
        doppler_scale = IF_FREQ / CARRIER_FREQ
        lo_offset = doppler_hz * doppler_scale
        baseband_signal, intermediate_signals = downconvert_to_baseband(
            rx_signal, rf_sample_rate, lo_offset)
    else:
        baseband_signal = rx_signal
        intermediate_signals = {}

    # Matched filtering
    filtered = np.convolve(baseband_signal, rrc, mode='same')

    # Symbol timing recovery (energy-based)
    best_offset = 0
    max_energy = 0
    for test_offset in range(SAMPLES_PER_SYMBOL):
        test_symbols = filtered[test_offset::SAMPLES_PER_SYMBOL][:min(100, num_symbols)]
        if len(test_symbols) > 10:
            energy = np.mean(np.abs(test_symbols)**2)
            if energy > max_energy:
                max_energy = energy
                best_offset = test_offset

    rx_symbols_raw = filtered[best_offset::SAMPLES_PER_SYMBOL][:num_symbols]

    # AGC / Power normalization
    rx_symbols_raw = rx_symbols_raw / np.sqrt(np.mean(np.abs(rx_symbols_raw)**2) + 1e-10)

    # Carrier phase and frequency recovery (symbol rate)
    rx_symbols = carrier_phase_recovery_symbols(rx_symbols_raw, loop_bw=0.02)

    # Demodulation and decoding
    rx_bits = qam64_demodulate(rx_symbols)
    descrambled = descramble(rx_bits)
    deinterleaved = block_deinterleave(descrambled)
    decoded = ldpc_decode(deinterleaved, code_rate=0.75)
    return decoded[:num_info_bits], rx_symbols, baseband_signal, intermediate_signals

# ============== LINK BUDGET ==============

def calculate_link_budget(distance_km):
    """Calculate link budget for 64-QAM MEO ISL"""
    distance = distance_km * 1000
    tx_power_dbw = 10 * np.log10(TX_POWER_W)
    tx_power_eff = tx_power_dbw - OUTPUT_BACKOFF_DB
    eirp = tx_power_eff + TX_ANTENNA_GAIN_DBI - LINE_LOSS_DB
    fspl = free_space_path_loss_db(distance)
    rx_gain = RX_ANTENNA_GAIN_DBI - LINE_LOSS_DB
    g_over_t = rx_gain - 10 * np.log10(SYSTEM_NOISE_TEMP_K)
    c_n0 = eirp - fspl - 0.5 + g_over_t + 228.6
    eb_n0 = c_n0 - 10 * np.log10(BIT_RATE)
    return {'EIRP_dBW': eirp, 'FSPL_dB': fspl, 'C_N0_dBHz': c_n0, 'Eb_N0_dB': eb_n0|}

# ============== BER CALCULATION ==============

def theoretical_64qam_ber(eb_n0_db):
    """Theoretical 64-QAM BER in AWGN"""
    from scipy.special import erfc
    eb_n0 = 10 ** (eb_n0_db / 10)
    es_n0 = 6 * eb_n0  # 6 bits per symbol
    return (7/24) * erfc(np.sqrt(es_n0 / 42))

def meo_64qam_isl_simulation(num_bits=60000, distance_km=10000, doppler_hz=60000,
                              enable_rf_simulation=True):
    """
    Complete MEO 64-QAM ISL system simulation with RF up/down conversion
    """
    print(f"MEO 64-QAM ISL Simulation")
    print(f"Carrier: {CARRIER_FREQ/1e9:.1f} GHz, Doppler: {doppler_hz/1000:.1f} kHz")
    print(f"RF Simulation: {'Enabled' if enable_rf_simulation else 'Disabled'}")
    if enable_rf_simulation:
        print(f"IF: {IF_FREQ/1e6:.1f} MHz, BB Rate: {BASEBAND_SAMPLE_RATE/1e6:.1f} MHz, RF Rate: {RF_SAMPLE_RATE/1e6:.1f} MHz")

    budget = calculate_link_budget(distance_km)
    print(f"Link Budget: EIRP={budget['EIRP_dBW']:.1f} dBW, Eb/N0={budget['Eb_N0_dB']:.1f} dB")

    num_info_bits = int(num_bits * 0.75)
    num_info_bits = (num_info_bits // 6) * 6
    tx_bits = np.random.randint(0, 2, num_info_bits)

    # TRANSMITTER with RF Up-Conversion
    tx_signal, tx_symbols, rrc, tx_sample_rate, tx_baseband = meo_64qam_transmitter(
        tx_bits, enable_rf_upconversion=enable_rf_simulation)

    # CHANNEL at 27 GHz RF
    if enable_rf_simulation:
        rx_signal = meo_rf_space_channel(tx_signal, tx_sample_rate, distance_km,
                                          doppler_hz, budget['Eb_N0_dB'])
    else:
        rx_signal = meo_space_channel(tx_signal, distance_km, doppler_hz, budget['Eb_N0_dB'])

    # RECEIVER with RF Down-Conversion
    num_symbols = len(tx_symbols)
    rx_bits, rx_symbols, rx_baseband, intermediate_signals = meo_64qam_receiver(
        rx_signal, rrc, num_symbols, num_info_bits, doppler_hz,
        rf_input=enable_rf_simulation, rf_sample_rate=tx_sample_rate)

    # BER Calculation
    min_len = min(len(tx_bits), len(rx_bits))
    bit_errors = np.sum(tx_bits[:min_len] != rx_bits[:min_len])
    ber = bit_errors / min_len

    print(f"Results: {len(tx_symbols)} symbols, {bit_errors} errors, BER={ber:.2e}")
    return {'tx_bits': tx_bits, 'rx_bits': rx_bits, 'tx_symbols': tx_symbols,
            'rx_symbols': rx_symbols, 'tx_baseband': tx_baseband, 'rx_baseband': rx_baseband,
            'ber': ber, 'link_budget': budget, 'intermediate_signals': intermediate_signals|}

64-QAM Space Environment Considerations
  • Maximum Throughput: 6 bits/symbol, 50% more than 16-QAM
  • High PAPR: 3.7 dB requires ~8 dB amplifier back-off
  • CFR Required: Crest factor reduction for PAPR control
  • Phase Noise Critical: <0.5 deg RMS phase error tolerance
  • I/Q Balance: <0.1 dB gain, <0.3 deg phase imbalance
  • DVB-S2X Compatible: Highest standard QAM mode
RF Up/Down Conversion Implementation
  • TX Upconversion: 10x oversample, LPF, quadrature mix to IF
  • IF Frequency: 100 MHz simulation representing 27 GHz
  • RX Downconversion: Quadrature demod with Doppler-adjusted LO
  • Doppler Comp: LO offset scaled by IF/Carrier ratio
  • Timing Recovery: Energy-based optimal sampling point search
  • Carrier Recovery: 2nd-order DD-PLL at symbol rate (0.02 BW)
