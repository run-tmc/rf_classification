"""
Broadcast FM MEO Inter-Satellite Link (ISL) System Implementation
Space-based signal processing chain for Medium Earth Orbit communications
Frequency Modulation with Pre/De-emphasis and FM Discriminator
"""

import numpy as np
from scipy import signal
from scipy.signal import butter, filtfilt, lfilter, firwin
from scipy.constants import c, k  # speed of light, Boltzmann constant

# ============== MEO ORBITAL CONSTANTS ==============
EARTH_RADIUS = 6371e3           # meters
EARTH_MU = 3.986e14             # gravitational parameter m^3/s^2
MEO_ALTITUDE = 20200e3          # 20,200 km (GPS orbit)

# ============== FM SYSTEM PARAMETERS ==============
CARRIER_FREQ = 2.2e9            # 2.2 GHz S-band carrier
WAVELENGTH = c / CARRIER_FREQ
BASEBAND_BW = 15e3              # 15 kHz max baseband frequency
FREQ_DEVIATION = 75e3           # +/-75 kHz frequency deviation
MOD_INDEX = FREQ_DEVIATION / BASEBAND_BW  # beta = 5.0
CARSON_BW = 2 * (FREQ_DEVIATION + BASEBAND_BW)  # 180 kHz

# Sampling rates
FS_BASEBAND = 48e3              # 48 kHz baseband sample rate
FS_IF = 480e3                   # 480 kHz IF sample rate (10x interpolation)
IF_FREQ = 70e3                  # 70 kHz intermediate frequency

# Link budget parameters
TX_POWER_W = 10                  # 10W SSPA
TX_ANTENNA_GAIN_DBI = 32        # 1.0m dish @ 2.2 GHz
RX_ANTENNA_GAIN_DBI = 32
SYSTEM_NOISE_TEMP_K = 300       # Kelvin
LINE_LOSS_DB = 1.0

# Pre/De-emphasis time constant (75 microseconds for broadcast FM)
TAU_EMPHASIS = 75e-6            # 75 us time constant


# ============== BASEBAND SOURCE ==============

def generate_baseband_signal(duration, fs=FS_BASEBAND, signal_type='tone'):
    """
    Generate baseband test signal for FM modulation

    Parameters:
    -----------
    duration : float - Signal duration in seconds
    fs : float - Sample rate in Hz
    signal_type : str - 'tone', 'multitone', or 'noise'

    Returns:
    --------
    t : ndarray - Time vector
    m : ndarray - Baseband message signal (normalized to +/-1)
    """
    t = np.arange(0, duration, 1/fs)

    if signal_type == 'tone':
        # Single 1 kHz tone
        m = np.sin(2 * np.pi * 1000 * t)
    elif signal_type == 'multitone':
        # Multi-tone test signal (sum of frequencies)
        freqs = [400, 1000, 3000, 7500, 12000]
        m = np.sum([np.sin(2 * np.pi * f * t) for f in freqs], axis=0)
        m = m / np.max(np.abs(m))  # Normalize
    elif signal_type == 'noise':
        # Band-limited noise (simulates audio)
        m = np.random.randn(len(t))
        b, a = butter(6, BASEBAND_BW/(0.5*fs), btype='low')
        m = filtfilt(b, a, m)
        m = m / np.max(np.abs(m))
    else:
        raise ValueError(f"Unknown signal type: {signal_type}")

    return t, m


# ============== PRE-EMPHASIS / DE-EMPHASIS FILTERS ==============

def pre_emphasis_filter(signal_in, fs, tau=TAU_EMPHASIS):
    """
    Apply pre-emphasis filter to boost high frequencies
    H(s) = 1 + s*tau  (high-frequency boost)
    """
    alpha = tau * fs
    b = np.array([1 + alpha, -alpha])
    a = np.array([1.0])
    signal_out = lfilter(b, a, signal_in)
    signal_out = signal_out / np.max(np.abs(signal_out))
    return signal_out


def de_emphasis_filter(signal_in, fs, tau=TAU_EMPHASIS):
    """
    Apply de-emphasis filter to attenuate high frequencies
    H(s) = 1 / (1 + s*tau)  (low-pass characteristic)
    """
    alpha = np.exp(-1 / (tau * fs))
    b = np.array([1 - alpha])
    a = np.array([1, -alpha])
    signal_out = lfilter(b, a, signal_in)
    return signal_out


# ============== FM MODULATION ==============

def fm_modulate(message, fs, fc=IF_FREQ, freq_dev=FREQ_DEVIATION):
    """
    FM modulation using phase accumulator method
    s(t) = A * cos(2*pi*fc*t + 2*pi*kf*integral(m(t)))

    Parameters:
    -----------
    message : ndarray - Baseband message signal (normalized to +/-1)
    fs : float - Sample rate
    fc : float - Carrier/IF frequency
    freq_dev : float - Maximum frequency deviation (Hz)

    Returns:
    --------
    fm_signal : ndarray - FM modulated signal
    inst_freq : ndarray - Instantaneous frequency
    """
    # Calculate instantaneous frequency
    inst_freq = fc + freq_dev * message

    # Phase is integral of instantaneous frequency
    phase = 2 * np.pi * np.cumsum(inst_freq) / fs

    # Generate FM signal
    fm_signal = np.cos(phase)

    return fm_signal, inst_freq


def nco_generate(freq, fs, num_samples, phase_offset=0):
    """
    Numerically Controlled Oscillator (NCO) / DDS
    """
    if np.isscalar(freq):
        t = np.arange(num_samples) / fs
        phase = 2 * np.pi * freq * t + phase_offset
    else:
        phase = 2 * np.pi * np.cumsum(freq) / fs + phase_offset
    nco_out = np.exp(1j * phase)
    return nco_out


# ============== FM DEMODULATION ==============

def fm_demodulate_diff(fm_signal, fs, freq_dev=FREQ_DEVIATION):
    """
    FM demodulation using differentiate-and-envelope method
    The instantaneous frequency is the derivative of the phase
    """
    # Convert to analytic signal using Hilbert transform
    analytic = signal.hilbert(fm_signal)

    # Extract instantaneous phase
    inst_phase = np.unwrap(np.angle(analytic))

    # Differentiate to get instantaneous frequency
    inst_freq = np.diff(inst_phase) * fs / (2 * np.pi)
    inst_freq = np.append(inst_freq, inst_freq[-1])

    # Remove DC and normalize
    demod = inst_freq - np.mean(inst_freq)
    demod = demod / freq_dev

    return demod


def fm_demodulate_pll(fm_signal, fs, fc=IF_FREQ, freq_dev=FREQ_DEVIATION,
                       loop_bw=5000, damping=0.707):
    """
    FM demodulation using Phase-Locked Loop (PLL)
    The PLL tracks the instantaneous phase of the FM signal
    """
    wn = 2 * np.pi * loop_bw / (damping + 1/(4*damping))
    K0, Kd = 1.0, 1.0
    tau1 = K0 * Kd / (wn * wn)
    tau2 = 2 * damping / wn

    phase_estimate = 0
    freq_estimate = 2 * np.pi * fc
    integrator = 0
    demod = np.zeros(len(fm_signal))

    for n in range(len(fm_signal)):
        lo_q = -np.sin(phase_estimate)
        phase_error = fm_signal[n] * lo_q
        integrator += phase_error / tau1 / fs
        loop_filter_out = integrator + phase_error * tau2 / tau1
        freq_estimate = 2 * np.pi * fc + K0 * loop_filter_out
        phase_estimate += freq_estimate / fs
        demod[n] = loop_filter_out

    demod = demod / (2 * np.pi * freq_dev)
    return demod


# ============== SIGNAL CONDITIONING ==============

def hard_limiter(signal_in):
    """Hard limiter - removes amplitude variations"""
    return np.sign(signal_in)


def interpolate_signal(signal_in, factor):
    """Interpolate signal by integer factor"""
    signal_up = np.zeros(len(signal_in) * factor)
    signal_up[::factor] = signal_in
    num_taps = 64 * factor + 1
    h = firwin(num_taps, 1.0/factor)
    signal_out = np.convolve(signal_up, h * factor, mode='same')
    return signal_out


def decimate_signal(signal_in, factor):
    """Decimate signal by integer factor"""
    num_taps = 64 * factor + 1
    h = firwin(num_taps, 1.0/factor)
    signal_filtered = np.convolve(signal_in, h, mode='same')
    signal_out = signal_filtered[::factor]
    return signal_out


# ============== SPACE CHANNEL MODEL ==============

def free_space_path_loss_db(distance_m, freq_hz=CARRIER_FREQ):
    """Calculate free space path loss in dB"""
    fspl = (4 * np.pi * distance_m * freq_hz / c) ** 2
    return 10 * np.log10(fspl)


def doppler_shift_hz(relative_velocity, freq_hz=CARRIER_FREQ):
    """Calculate Doppler frequency shift"""
    return freq_hz * relative_velocity / c


def apply_doppler(signal_in, doppler_hz, fs):
    """Apply Doppler frequency shift to signal"""
    t = np.arange(len(signal_in)) / fs
    shift = np.exp(1j * 2 * np.pi * doppler_hz * t)
    return np.real(signal_in * shift)


def add_awgn(signal_in, snr_db, signal_power=None):
    """Add Additive White Gaussian Noise"""
    if signal_power is None:
        signal_power = np.mean(signal_in ** 2)
    snr_linear = 10 ** (snr_db / 10)
    noise_power = signal_power / snr_linear
    noise = np.sqrt(noise_power) * np.random.randn(len(signal_in))
    return signal_in + noise


# ============== SNR AND QUALITY METRICS ==============

def calculate_snr_db(signal_clean, signal_noisy):
    """Calculate SNR between clean and noisy signals"""
    noise = signal_noisy - signal_clean
    signal_power = np.mean(signal_clean ** 2)
    noise_power = np.mean(noise ** 2)
    if noise_power == 0:
        return np.inf
    return 10 * np.log10(signal_power / noise_power)


def fm_improvement_factor_db(mod_index):
    """Calculate FM improvement factor: 3 * beta^2 * (beta + 1)"""
    improvement = 3 * mod_index ** 2 * (mod_index + 1)
    return 10 * np.log10(improvement)


# ============== COMPLETE FM TRANSCEIVER ==============

class FMTransmitter:
    """FM Transmitter for MEO Space ISL"""

    def __init__(self, fs_baseband=FS_BASEBAND, fs_if=FS_IF,
                 freq_dev=FREQ_DEVIATION, if_freq=IF_FREQ):
        self.fs_baseband = fs_baseband
        self.fs_if = fs_if
        self.freq_dev = freq_dev
        self.if_freq = if_freq
        self.interp_factor = int(fs_if / fs_baseband)

    def transmit(self, message):
        """Complete FM transmit chain"""
        msg_preemph = pre_emphasis_filter(message, self.fs_baseband)
        msg_limited = np.clip(msg_preemph, -1, 1)
        msg_interp = interpolate_signal(msg_limited, self.interp_factor)
        fm_signal, _ = fm_modulate(msg_interp, self.fs_if,
                                   self.if_freq, self.freq_dev)
        return fm_signal


class FMReceiver:
    """FM Receiver for MEO Space ISL"""

    def __init__(self, fs_baseband=FS_BASEBAND, fs_if=FS_IF,
                 freq_dev=FREQ_DEVIATION, if_freq=IF_FREQ,
                 demod_method='diff'):
        self.fs_baseband = fs_baseband
        self.fs_if = fs_if
        self.freq_dev = freq_dev
        self.if_freq = if_freq
        self.decim_factor = int(fs_if / fs_baseband)
        self.demod_method = demod_method

    def receive(self, fm_signal):
        """Complete FM receive chain"""
        fm_limited = hard_limiter(fm_signal)

        if self.demod_method == 'diff':
            demod = fm_demodulate_diff(fm_limited, self.fs_if, self.freq_dev)
        elif self.demod_method == 'pll':
            demod = fm_demodulate_pll(fm_limited, self.fs_if,
                                      self.if_freq, self.freq_dev)
        else:
            raise ValueError(f"Unknown demod method: {self.demod_method}")

        demod_dec = decimate_signal(demod, self.decim_factor)
        demod_deemph = de_emphasis_filter(demod_dec, self.fs_baseband)
        return demod_deemph


def meo_fm_isl_simulation(duration=0.1, snr_db=30, doppler_hz=5000):
    """Complete MEO FM Inter-Satellite Link simulation"""
    t, message = generate_baseband_signal(duration, signal_type='multitone')

    tx = FMTransmitter()
    rx = FMReceiver(demod_method='diff')

    fm_tx = tx.transmit(message)
    fm_doppler = apply_doppler(fm_tx, doppler_hz, FS_IF)
    fm_noisy = add_awgn(fm_doppler, snr_db)
    demod = rx.receive(fm_noisy)

    delay = len(message) - len(demod)
    if delay > 0:
        message_aligned = message[delay//2:-delay//2 if delay//2 > 0 else None]
        demod_aligned = demod
    else:
        message_aligned = message
        demod_aligned = demod[:len(message)]

    output_snr = calculate_snr_db(message_aligned, demod_aligned)
    fm_improvement = fm_improvement_factor_db(MOD_INDEX)

    results = {
        'time': t,
        'message': message,
        'demodulated': demod,
        'input_snr_db': snr_db,
        'output_snr_db': output_snr,
        'fm_improvement_db': fm_improvement,
        'mod_index': MOD_INDEX,
        'carson_bw_hz': CARSON_BW
    }
    return results
