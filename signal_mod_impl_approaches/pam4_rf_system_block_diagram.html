"""
PAM4 RF Transmitter-Receiver System Implementation
Complete signal processing chain for 4-level pulse amplitude modulation
"""

import numpy as np
from scipy import signal
from scipy.signal import butter, filtfilt

# ============== SYSTEM PARAMETERS ==============
SYMBOL_RATE = 1e6          # 1 MSym/s
SAMPLES_PER_SYMBOL = 8      # Oversampling factor
SAMPLE_RATE = SYMBOL_RATE * SAMPLES_PER_SYMBOL
CARRIER_FREQ = 100e6        # 100 MHz carrier
ROLLOFF = 0.35              # RRC roll-off factor
FILTER_SPAN = 10             # Filter span in symbols

# PAM4 symbol mapping (Gray coded)
PAM4_LEVELS = np.array([-3, -1, 1, 3])
PAM4_MAP = {(0,0): -3, (0,1): -1, (1,0): 1, (1,1): 3}

# ============== TRANSMITTER FUNCTIONS ==============

def generate_random_bits(num_bits):
    """Generate random binary data source"""
    return np.random.randint(0, 2, num_bits)

def bits_to_pam4_symbols(bits):
    """Map bit pairs to PAM4 symbols using Gray coding"""
    # Ensure even number of bits
    if len(bits) % 2 != 0:
        bits = np.append(bits, 0)

    # Reshape into pairs and map to symbols
    bit_pairs = bits.reshape(-1, 2)
    symbols = np.array([PAM4_MAP[tuple(pair)] for pair in bit_pairs])
    return symbols

def rrc_filter(num_taps, alpha, sps):
    """Design Root Raised Cosine pulse shaping filter"""
    t = np.arange(-num_taps//2, num_taps//2 + 1) / sps
    h = np.zeros_like(t, dtype=float)

    for i, ti in enumerate(t):
        if ti == 0:
            h[i] = 1 + alpha * (4/np.pi - 1)
        elif abs(ti) == 1/(4*alpha):
            h[i] = alpha/np.sqrt(2) * ((1+2/np.pi)*np.sin(np.pi/(4*alpha)) +
                                         (1-2/np.pi)*np.cos(np.pi/(4*alpha)))
        else:
            num = np.sin(np.pi*ti*(1-alpha)) + 4*alpha*ti*np.cos(np.pi*ti*(1+alpha))
            den = np.pi*ti*(1 - (4*alpha*ti)**2)
            h[i] = num / den

    return h / np.sqrt(np.sum(h**2))  # Normalize

def upsample_and_filter(symbols, sps, rrc):
    """Upsample symbols and apply pulse shaping"""
    # Insert zeros for upsampling
    upsampled = np.zeros(len(symbols) * sps)
    upsampled[::sps] = symbols

    # Apply RRC filter
    shaped = np.convolve(upsampled, rrc, mode='same')
    return shaped

def rf_upconvert(baseband, carrier_freq, sample_rate):
    """Upconvert baseband signal to RF carrier"""
    t = np.arange(len(baseband)) / sample_rate
    carrier = np.cos(2 * np.pi * carrier_freq * t)
    rf_signal = baseband * carrier
    return rf_signal

# ============== CHANNEL FUNCTIONS ==============

def awgn_channel(signal, snr_db):
    """Add white Gaussian noise to achieve target SNR"""
    signal_power = np.mean(signal**2)
    snr_linear = 10**(snr_db/10)
    noise_power = signal_power / snr_linear
    noise = np.sqrt(noise_power) * np.random.randn(len(signal))
    return signal + noise

def multipath_channel(signal, taps=[1.0, 0.3, 0.1]):
    """Apply multipath fading channel"""
    return np.convolve(signal, taps, mode='same')

# ============== RECEIVER FUNCTIONS ==============

def rf_downconvert(rf_signal, carrier_freq, sample_rate, cutoff_ratio=0.2):
    """Downconvert RF signal to baseband"""
    t = np.arange(len(rf_signal)) / sample_rate
    lo = np.cos(2 * np.pi * carrier_freq * t)
    mixed = rf_signal * lo * 2  # Factor of 2 to recover amplitude

    # Low-pass filter to remove 2*fc component
    nyq = sample_rate / 2
    cutoff = cutoff_ratio * nyq
    b, a = butter(5, cutoff/nyq, btype='low')
    baseband = filtfilt(b, a, mixed)
    return baseband

def matched_filter(signal, rrc):
    """Apply matched filter (same as TX RRC)"""
    return np.convolve(signal, rrc, mode='same')

def timing_recovery_downsample(signal, sps, offset=0):
    """Downsample to 1 sample per symbol with timing offset"""
    return signal[offset::sps]

def agc_normalize(signal):
    """Automatic gain control - normalize signal amplitude"""
    return signal / np.std(signal) * np.sqrt(5)  # Scale to match PAM4 variance

def pam4_slicer(symbols):
    """Decision device - slice to nearest PAM4 level"""
    thresholds = [-2, 0, 2]
    indices = np.digitize(symbols, thresholds)
    return PAM4_LEVELS[indices]

def pam4_symbols_to_bits(symbols):
    """Demap PAM4 symbols back to bits"""
    DEMAP = {-3: [0,0], -1: [0,1], 1: [1,0], 3: [1,1]}
    bits = np.array([DEMAP[int(s)] for s in symbols]).flatten()
    return bits

def calculate_ber(tx_bits, rx_bits):
    """Calculate Bit Error Rate"""
    min_len = min(len(tx_bits), len(rx_bits))
    errors = np.sum(tx_bits[:min_len] != rx_bits[:min_len])
    return errors / min_len

# ============== MAIN SIMULATION ==============

def pam4_rf_system(num_symbols=1000, snr_db=20):
    """Complete PAM4 RF TX/RX system simulation"""

    # Design RRC filter
    num_taps = FILTER_SPAN * SAMPLES_PER_SYMBOL + 1
    rrc = rrc_filter(num_taps, ROLLOFF, SAMPLES_PER_SYMBOL)

    # ===== TRANSMITTER =====
    tx_bits = generate_random_bits(num_symbols * 2)
    tx_symbols = bits_to_pam4_symbols(tx_bits)
    tx_shaped = upsample_and_filter(tx_symbols, SAMPLES_PER_SYMBOL, rrc)
    tx_rf = rf_upconvert(tx_shaped, CARRIER_FREQ, SAMPLE_RATE)

    # ===== CHANNEL =====
    rx_rf = awgn_channel(tx_rf, snr_db)

    # ===== RECEIVER =====
    rx_baseband = rf_downconvert(rx_rf, CARRIER_FREQ, SAMPLE_RATE)
    rx_filtered = matched_filter(rx_baseband, rrc)
    rx_normalized = agc_normalize(rx_filtered)
    rx_samples = timing_recovery_downsample(rx_normalized, SAMPLES_PER_SYMBOL,
                                            offset=num_taps//2)
    rx_symbols = pam4_slicer(rx_samples[:num_symbols])
    rx_bits = pam4_symbols_to_bits(rx_symbols)

    # Calculate BER
    ber = calculate_ber(tx_bits, rx_bits)

    return {
        'tx_bits': tx_bits,
        'tx_symbols': tx_symbols,
        'rx_symbols': rx_symbols,
        'rx_bits': rx_bits,
        'ber': ber
    }

# Run simulation
if __name__ == "__main__":
    results = pam4_rf_system(num_symbols=10000, snr_db=25)
    print(f"Bit Error Rate: {results['ber']:.6f}")
Key Design Considerations
  • Gray Coding: Minimizes bit errors - adjacent symbols differ by 1 bit
  • RRC Filter: TX and RX filters combine to form Raised Cosine (zero ISI)
  • Timing Recovery: Critical for correct symbol sampling instant
  • AGC: Normalizes signal before slicing to handle varying channel gain
  • Decision Thresholds: Optimal at midpoints: -2, 0, +2
