"""
PAM4 MEO Inter-Satellite Link (ISL) System Implementation
Space-based signal processing chain for Medium Earth Orbit communications
Includes: LDPC FEC, Doppler compensation, space channel model
"""

import numpy as np
import matplotlib.pyplot as plt
from scipy import signal
from scipy.signal import butter, filtfilt, resample_poly
from scipy.constants import c, k  # speed of light, Boltzmann
import pickle
import os

# ============== MEO ORBITAL CONSTANTS ==============
EARTH_RADIUS = 6371e3           # meters
EARTH_MU = 3.986e14             # gravitational parameter m^3/s^2
MEO_ALTITUDE = 20200e3          # 20,200 km (GPS orbit)
MEO_RADIUS = EARTH_RADIUS + MEO_ALTITUDE

# ============== RF SYSTEM PARAMETERS ==============
CARRIER_FREQ = 23e9             # 23 GHz Ka-band
WAVELENGTH = c / CARRIER_FREQ
SYMBOL_RATE = 100e6             # 100 MSym/s
SAMPLES_PER_SYMBOL = 8
SAMPLE_RATE = SYMBOL_RATE * SAMPLES_PER_SYMBOL
ROLLOFF = 0.25                  # Tighter rolloff for space
FILTER_SPAN = 12                 # symbols

# Link budget parameters
TX_POWER_W = 20                  # 20W TWTA
TX_ANTENNA_GAIN_DBI = 40        # 0.5m dish @ 23 GHz
RX_ANTENNA_GAIN_DBI = 40
SYSTEM_NOISE_TEMP_K = 500       # Kelvin
LINE_LOSS_DB = 1.0

# PAM4 symbol mapping (Gray coded)
PAM4_LEVELS = np.array([-3, -1, 1, 3])
PAM4_MAP = {(0,0): -3, (0,1): -1, (1,0): 1, (1,1): 3}
PAM4_DEMAP = {-3: (0,0), -1: (0,1), 1: (1,0), 3: (1,1)}

# ============== ORBITAL MECHANICS ==============

def meo_orbital_velocity(altitude=MEO_ALTITUDE):
    """Calculate orbital velocity at given altitude"""
    r = EARTH_RADIUS + altitude
    return np.sqrt(EARTH_MU / r)  # m/s

def max_doppler_shift(relative_velocity, carrier_freq=CARRIER_FREQ):
    """Calculate maximum Doppler shift for given relative velocity"""
    return (relative_velocity / c) * carrier_freq

def isl_range(sat1_pos, sat2_pos):
    """Calculate inter-satellite link distance"""
    return np.linalg.norm(sat2_pos - sat1_pos)

def propagation_delay(distance):
    """Calculate one-way propagation delay"""
    return distance / c

# ============== SPACE CHANNEL MODEL ==============

def free_space_path_loss_db(distance, freq=CARRIER_FREQ):
    """Calculate FSPL in dB: FSPL = 20*log10(4*pi*d*f/c)"""
    fspl = (4 * np.pi * distance * freq / c) ** 2
    return 10 * np.log10(fspl)

def apply_doppler_shift(signal_in, doppler_hz, sample_rate):
    """Apply Doppler frequency shift to signal"""
    t = np.arange(len(signal_in)) / sample_rate
    shift = np.exp(2j * np.pi * doppler_hz * t)
    return signal_in * shift

def space_thermal_noise(signal_len, bandwidth, noise_temp=SYSTEM_NOISE_TEMP_K):
    """Generate thermal noise for space environment"""
    noise_power = k * noise_temp * bandwidth  # Watts
    noise_std = np.sqrt(noise_power)
    return noise_std * (np.random.randn(signal_len) + 1j*np.random.randn(signal_len)) / np.sqrt(2)

def pointing_loss_db(pointing_error_deg, beamwidth_deg):
    """Calculate antenna pointing loss"""
    # Approximate Gaussian beam pattern
    return 12 * (pointing_error_deg / beamwidth_deg) ** 2

def phase_noise(signal_len, phase_noise_std_rad):
    """Generate phase noise (random walk model)"""
    phase_increments = phase_noise_std_rad * np.random.randn(signal_len)
    phase = np.cumsum(phase_increments)
    return np.exp(1j * phase)

def meo_space_channel(signal_in, distance_km, doppler_hz, snr_db):
    """Complete MEO space channel model

    Note: FSPL and antenna gains are already accounted for in the link budget
    calculation that produces snr_db. Applying them here would double-count
    the path loss and result in unrealistically low SNR.
    """
    # Apply Doppler shift
    sig = apply_doppler_shift(signal_in, doppler_hz, SAMPLE_RATE)

    # Apply phase noise (oscillator instability)
    sig = sig * phase_noise(len(sig), 0.01)  # ~0.01 rad RMS

    # Add AWGN based on SNR (FSPL already accounted for in link budget)
    signal_power = np.mean(np.abs(sig) ** 2)
    snr_linear = 10 ** (snr_db / 10)
    noise_power = signal_power / snr_linear
    noise = np.sqrt(noise_power/2) * (np.random.randn(len(sig)) + 1j*np.random.randn(len(sig)))

    return sig + noise

# ============== FEC: LDPC (Simplified) ==============

def ldpc_encode(bits, code_rate=0.75):
    """Simplified Low-Density Parity-Check (LDPC) encoder (adds parity bits)"""
    # In practice, use proper LDPC from libraries like sionna or aff3ct
    k = len(bits)
    n = int(k / code_rate)
    parity_len = n - k

    # Simple parity check (placeholder for real LDPC)
    parity = np.zeros(parity_len, dtype=int)
    for i in range(parity_len):
        parity[i] = np.sum(bits[i::parity_len]) % 2

    return np.concatenate([bits, parity])

def ldpc_decode(bits, code_rate=0.75):
    """Simplified LDPC decoder (extracts data bits)"""
    k = int(len(bits) * code_rate)
    return bits[:k]

# ============== INTERLEAVER ==============

def block_interleave(bits, rows=64):
    """Block interleaver for burst error protection"""
    cols = int(np.ceil(len(bits) / rows))
    padded = np.zeros(rows * cols, dtype=int)
    padded[:len(bits)] = bits

    matrix = padded.reshape(rows, cols)
    return matrix.T.flatten()

def block_deinterleave(bits, rows=64):
    """Block de-interleaver"""
    cols = int(np.ceil(len(bits) / rows))
    required_size = rows * cols
    # Pad if input is shorter than required (due to edge effects in receiver)
    if len(bits) < required_size:
        bits = np.concatenate([bits, np.zeros(required_size - len(bits), dtype=int)])
    matrix = bits[:required_size].reshape(cols, rows)
    return matrix.T.flatten()

# ============== SCRAMBLER ==============

def lfsr_sequence(length, seed=0x1ACFFC1D):
    """Generate Linear Feedback Shift Register (LFSR) pseudo-random sequence"""
    state = seed
    seq = np.zeros(length, dtype=int)
    for i in range(length):
        seq[i] = state & 1
        newbit = ((state >> 0) ^ (state >> 2) ^ (state >> 3) ^ (state >> 5)) & 1
        state = (state >> 1) | (newbit << 31)
    return seq

def scramble(bits):
    """Scramble bits with LFSR sequence"""
    pn = lfsr_sequence(len(bits))
    return (bits ^ pn).astype(int)

def descramble(bits):
    """Descramble bits (same operation as scramble)"""
    return scramble(bits)

# ============== TRANSMITTER ==============

def rrc_filter(num_taps, alpha, sps):
    """Root Raised Cosine filter design"""
    t = np.arange(-num_taps//2, num_taps//2 + 1) / sps
    h = np.zeros_like(t, dtype=float)

    for i, ti in enumerate(t):
        if ti == 0:
            h[i] = 1 + alpha * (4/np.pi - 1)
        elif abs(abs(ti) - 1/(4*alpha)) < 1e-10:
            h[i] = alpha/np.sqrt(2) * ((1+2/np.pi)*np.sin(np.pi/(4*alpha)) +
                                         (1-2/np.pi)*np.cos(np.pi/(4*alpha)))
        else:
            num = np.sin(np.pi*ti*(1-alpha)) + 4*alpha*ti*np.cos(np.pi*ti*(1+alpha))
            den = np.pi*ti*(1 - (4*alpha*ti)**2)
            h[i] = num / (den + 1e-10)

    return h / np.sqrt(np.sum(h**2))

def bits_to_pam4(bits):
    """Map bits to PAM4 symbols"""
    if len(bits) % 2 != 0:
        bits = np.append(bits, 0)
    bit_pairs = bits.reshape(-1, 2)
    symbols = np.array([PAM4_MAP[tuple(pair)] for pair in bit_pairs], dtype=float)
    return symbols

def twta_predistort(signal_in, backoff_db=3):
    """Predistortion for TWTA nonlinearity"""
    # Saleh model inverse (simplified)
    return signal_in * 10 ** (-backoff_db / 20)

def meo_transmitter(data_bits):
    """Complete MEO satellite transmitter chain

    Returns complex baseband signal (no IF upconversion).
    This ensures consistency with the complex baseband channel model.
    """
    # FEC encoding
    encoded = ldpc_encode(data_bits)

    # Interleaving
    interleaved = block_interleave(encoded)

    # Scrambling
    scrambled = scramble(interleaved)

    # PAM4 mapping
    symbols = bits_to_pam4(scrambled)

    # Upsampling
    upsampled = np.zeros(len(symbols) * SAMPLES_PER_SYMBOL, dtype=complex)
    upsampled[::SAMPLES_PER_SYMBOL] = symbols

    # Pulse shaping
    num_taps = FILTER_SPAN * SAMPLES_PER_SYMBOL + 1
    rrc = rrc_filter(num_taps, ROLLOFF, SAMPLES_PER_SYMBOL)
    shaped = np.convolve(upsampled, rrc, mode='same')

    # Predistortion for TWTA
    predistorted = twta_predistort(shaped)

    # Output complex baseband signal (no IF upconversion)
    # Channel model uses complex baseband Doppler representation
    return predistorted, symbols, rrc

# ============== RECEIVER ==============

def doppler_estimate_fft(signal_in, search_range_hz=100000):
    """Coarse Doppler estimation using FFT"""
    fft_size = 2 ** int(np.ceil(np.log2(len(signal_in))))
    spectrum = np.fft.fftshift(np.abs(np.fft.fft(signal_in, fft_size)))
    freqs = np.fft.fftshift(np.fft.fftfreq(fft_size, 1/SAMPLE_RATE))

    # Find peak in search range
    mask = np.abs(freqs) < search_range_hz
    peak_idx = np.argmax(spectrum * mask)
    return freqs[peak_idx]

def carrier_pll(signal_in, loop_bw=0.01):
    """Costas loop for carrier recovery"""
    phase = 0
    freq = 0
    output = np.zeros_like(signal_in, dtype=complex)

    # Loop filter coefficients
    damping = 0.707
    K1 = 4 * damping * loop_bw
    K2 = 4 * loop_bw ** 2

    for i in range(len(signal_in)):
        # Mix with NCO
        output[i] = signal_in[i] * np.exp(-1j * phase)

        # Phase detector (for PAM4)
        error = np.sign(np.real(output[i])) * np.imag(output[i])

        # Loop filter
        freq += K2 * error
        phase += K1 * error + freq

    return output

def gardner_ted(signal_in, sps):
    """Gardner timing error detector"""
    # Simplified: return signal at optimal sampling points
    mid = sps // 2
    timing_error = []

    for i in range(sps, len(signal_in) - sps, sps):
        # Gardner TED: e = Re{(y[n] - y[n-1]) * conj(y[n-1/2])}
        e = np.real((signal_in[i] - signal_in[i-sps]) * np.conj(signal_in[i-mid]))
        timing_error.append(e)

    return np.array(timing_error)

def pam4_slicer(symbols):
    """Hard decision PAM4 slicer"""
    thresholds = [-2, 0, 2]
    indices = np.digitize(np.real(symbols), thresholds)
    return PAM4_LEVELS[indices]

def pam4_to_bits(symbols):
    """Demap PAM4 symbols to bits"""
    bits = []
    for s in symbols:
        bits.extend(PAM4_DEMAP[int(s)])
    return np.array(bits)

def meo_receiver(baseband_signal, rrc, num_symbols, doppler_hz=0):
    """Complete MEO satellite receiver chain

    Accepts complex baseband signal (consistent with TX and channel model).
    """
    t = np.arange(len(baseband_signal)) / SAMPLE_RATE

    # Input is already complex baseband - no RF downconversion needed
    baseband_input = baseband_signal.copy()

    # Low-pass filter (removes out-of-band noise)
    nyq = SAMPLE_RATE / 2
    b, a = butter(5, SYMBOL_RATE / nyq, btype='low')
    baseband_after_lpf = filtfilt(b, a, baseband_input)

    # Doppler compensation (if known or estimated)
    if doppler_hz != 0:
        baseband_after_doppler = apply_doppler_shift(baseband_after_lpf, -doppler_hz, SAMPLE_RATE)
    else:
        baseband_after_doppler = baseband_after_lpf.copy()

    # Store intermediate signals for plotting
    intermediate_signals = {
        'after_downconv': baseband_input,  # Now just the input baseband
        'after_lpf': baseband_after_lpf,
        'after_doppler_comp': baseband_after_doppler,
        'time': t
    }

    # Carrier recovery (handles residual phase/frequency errors)
    recovered = carrier_pll(baseband_after_doppler)

    # Matched filter
    filtered = np.convolve(recovered, rrc, mode='same')

    # Downsample at optimal sampling instants
    # The combined TX RRC + RX RRC (raised cosine) has zero ISI at symbol centers
    # With mode='same' convolution, the group delay is minimal (empirically ~2 samples)
    offset = 2
    rx_symbols = filtered[offset::SAMPLES_PER_SYMBOL][:num_symbols]

    # AGC normalization
    rx_symbols = rx_symbols / np.std(np.real(rx_symbols)) * np.sqrt(5)

    # PAM4 slicing
    detected = pam4_slicer(rx_symbols)

    # Symbol to bits
    rx_bits = pam4_to_bits(detected)

    # Descramble
    descrambled = descramble(rx_bits)

    # De-interleave
    deinterleaved = block_deinterleave(descrambled)

    # LDPC decode
    decoded = ldpc_decode(deinterleaved)

    return decoded, rx_symbols, intermediate_signals


def plot_baseband_signals(intermediate_signals, num_samples=2000):

    Plot three signal level vs time plots showing baseband values after:

    2. Low-pass filtering

    ax1.set_title('After RF Downconversion', fontsize=11)
    ax1.legend(loc='upper right', fontsize=8)
    ax1.grid(True, alpha=0.3)
    ax1.set_xlim([t_plot[0], t_plot[-1]])

    # Plot 2: After Low-Pass Filtering
    ax2 = axes[1]
    signal_lpf = intermediate_signals['after_lpf'][:n]
    ax2.plot(t_plot, np.real(signal_lpf), 'b-', linewidth=0.5, alpha=0.8, label='I (Real)')

    ax2.set_title('After Low-Pass Filtering', fontsize=11)
    ax2.legend(loc='upper right', fontsize=8)
    ax2.grid(True, alpha=0.3)

    # Plot 3: After Doppler Compensation
    ax3 = axes[2]
    signal_doppler = intermediate_signals['after_doppler_comp'][:n]
    ax3.plot(t_plot, np.real(signal_doppler), 'b-', linewidth=0.5, alpha=0.8, label='I (Real)')

    ax3.set_xlabel('Time (microseconds)', fontsize=10)
    ax3.set_ylabel('Amplitude', fontsize=10)
    ax3.set_title('After Doppler Compensation', fontsize=11)
    ax3.legend(loc='upper right', fontsize=8)
    ax3.grid(True, alpha=0.3)

    plt.tight_layout()
    plt.savefig('baseband_signal_processing_stages.png', dpi=150, bbox_inches='tight')
    plt.show()

    print(f"\nBaseband signal plots saved to 'baseband_signal_processing_stages.png'")


def save_intermediate_signals(intermediate_signals, filename='intermediate_signals.pkl', output_dir=None):
    """
    Save the intermediate_signals dictionary to a pickle file.

    Parameters:
    -----------
    intermediate_signals : dict
        Dictionary containing intermediate signal data from the receiver chain
    filename : str
        Name of the output file (default: 'intermediate_signals.pkl')
    output_dir : str, optional
        Directory to save the file. If None, uses current working directory.

    Returns:
    --------
    str : Full path to the saved file
    """
    if output_dir is None:
        output_dir = os.getcwd()

    filepath = os.path.join(output_dir, filename)

    with open(filepath, 'wb') as f:
        pickle.dump(intermediate_signals, f)

    print(f"\nIntermediate signals saved to '{filepath}'")
    print(f"  Keys: {list(intermediate_signals.keys())}")
    print(f"  Signal lengths: {len(intermediate_signals['after_downconv'])} samples")

    return filepath


# ============== LINK BUDGET CALCULATOR ==============

def calculate_link_budget(distance_km):
    """Calculate complete link budget for MEO ISL"""
    distance = distance_km * 1000

    # Transmitter
    tx_power_dbw = 10 * np.log10(TX_POWER_W)
    eirp = tx_power_dbw + TX_ANTENNA_GAIN_DBI - LINE_LOSS_DB

    # Path loss
    fspl = free_space_path_loss_db(distance)
    pointing_loss = 0.5  # dB, typical

    # Receiver
    rx_gain = RX_ANTENNA_GAIN_DBI - LINE_LOSS_DB
    g_over_t = rx_gain - 10 * np.log10(SYSTEM_NOISE_TEMP_K)

    # C/N0
    c_n0 = eirp - fspl - pointing_loss + g_over_t + 228.6  # 228.6 = 10*log10(k)

    # Eb/N0
    bit_rate = SYMBOL_RATE * 2  # PAM4 = 2 bits/symbol
    eb_n0 = c_n0 - 10 * np.log10(bit_rate)

    return {
        'EIRP_dBW': eirp,
        'FSPL_dB': fspl,
        'C_N0_dBHz': c_n0,
        'Eb_N0_dB': eb_n0,
        'distance_km': distance_km
    }

# ============== MAIN SIMULATION ==============

def meo_pam4_isl_simulation(num_symbols=10000, distance_km=10000, doppler_hz=50000):
    """Complete MEO PAM4 ISL system simulation"""

    print(f"MEO PAM4 ISL Simulation")
    print(f"======================")
    print(f"Distance: {distance_km} km")
    print(f"Doppler: {doppler_hz/1000:.1f} kHz")
    print()

    # Calculate link budget
    budget = calculate_link_budget(distance_km)
    print(f"Link Budget:")
    print(f"  EIRP: {budget['EIRP_dBW']:.1f} dBW")
    print(f"  FSPL: {budget['FSPL_dB']:.1f} dB")
    print(f"  C/N0: {budget['C_N0_dBHz']:.1f} dBHz")
    print(f"  Eb/N0: {budget['Eb_N0_dB']:.1f} dB")
    print()

    # Generate random data
    num_data_bits = num_symbols * 2  # 2 bits per PAM4 symbol
    # Account for FEC overhead
    num_info_bits = int(num_data_bits * 0.75)  # Rate 3/4 LDPC
    tx_bits = np.random.randint(0, 2, num_info_bits)

    # Transmitter (outputs complex baseband)
    tx_baseband, tx_symbols, rrc = meo_transmitter(tx_bits)

    # Channel (use Eb/N0 from link budget)
    rx_baseband = meo_space_channel(tx_baseband, distance_km, doppler_hz, budget['Eb_N0_dB'])

    # Receiver (complex baseband input)
    rx_bits, rx_symbols, intermediate_signals = meo_receiver(rx_baseband, rrc, len(tx_symbols), doppler_hz)

    # Calculate BER
    min_len = min(len(tx_bits), len(rx_bits))
    bit_errors = np.sum(tx_bits[:min_len] != rx_bits[:min_len])
    ber = bit_errors / min_len

    print(f"Results:")
    print(f"  TX bits: {len(tx_bits)}")
    print(f"  RX bits: {len(rx_bits)}")
    print(f"  Bit errors: {bit_errors}")
    print(f"  BER: {ber:.2e}")

    # Plot baseband signal processing stages
    plot_baseband_signals(intermediate_signals)

    # Save intermediate signals to file
    save_intermediate_signals(intermediate_signals)

    return {
        'tx_bits': tx_bits,
        'rx_bits': rx_bits,
        'tx_symbols': tx_symbols,
        'rx_symbols': rx_symbols,
        'ber': ber,
        'link_budget': budget,
        'intermediate_signals': intermediate_signals
    }

MEO Space Environment Challenges
  • High Doppler: Up to ±75 kHz due to orbital velocities (~4 km/s)
  • Large FSPL: 200-220 dB path loss at Ka-band over MEO distances
  • Pointing Accuracy: <0.1° required for narrow beam antennas
  • Thermal Cycling: -150°C to +150°C during eclipse transitions
  • Radiation: Van Allen belt exposure requires rad-hard components
  • Propagation Delay: 3-150 ms one-way, affects ARQ protocols
  • Solar Noise: Increased noise temp when antenna points near Sun
Key Design Mitigations
  • LDPC FEC: Near-Shannon limit coding for 2+ dB coding gain
  • Interleaving: Distributes burst errors from radiation hits
  • AFC Loop: Tracks Doppler variations during pass
  • Predistortion: Compensates TWTA AM/AM and AM/PM
  • High-Gain Antennas: 40+ dBi to close long-range links
  • Low Noise LNA: Cryogenic or GaAs for NF < 2 dB
  • Robust Timing: Gardner TED handles SNR degradation
